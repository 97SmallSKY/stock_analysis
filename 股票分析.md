## 二、基础策略回测框架

### 核心逻辑

```
T日收盘后：龙虎榜公布 → 筛选净买入最大的股票
T+1日开盘：买入
T+1日收盘（或T+2开盘）：卖出
计算胜率和收益
```

```python
def calculate_net_buy(lhb_df):
    """
    计算每只股票每天的净买入额
    龙虎榜数据包含：买入前5和卖出前5的营业部
    """
    # 根据akshare返回的字段名调整
    # 典型字段：代码、名称、上榜日期、买入额、卖出额、净买入额
    
    # 按股票+日期汇总净买入
    net_buy = lhb_df.groupby(['代码', '上榜日期']).agg({
        '买入额': 'sum',
        '卖出额': 'sum',
        '净买入额': 'sum'
    }).reset_index()
    
    # 按日期分组，取净买入最大的
    net_buy['rank'] = net_buy.groupby('上榜日期')['净买入额'].rank(
        ascending=False, method='first'
    )
    
    return net_buy

def get_next_day_return(stock_code, trade_date, hold_days=1):
    """
    获取T+1日的收益率
    buy_price: T+1开盘价
    sell_price: T+1收盘价 或 T+2开盘价
    """
    try:
        # 标准化股票代码
        if stock_code.startswith('6'):
            symbol = f"sh{stock_code}"
        else:
            symbol = f"sz{stock_code}"
        
        # 获取日K线
        df = ak.stock_zh_a_hist(
            symbol=stock_code, 
            period="daily",
            start_date=trade_date,
            end_date=(pd.to_datetime(trade_date) + timedelta(days=10)).strftime('%Y%m%d'),
            adjust="qfq"
        )
        
        if df is None or len(df) < 2:
            return None
        
        # T+1日开盘买入
        buy_price = df.iloc[1]['开盘']  # 第二行是T+1日
        
        if hold_days == 1:
            # T+1日收盘卖出
            sell_price = df.iloc[1]['收盘']
        else:
            # T+N日收盘卖出
            if len(df) > 1 + hold_days:
                sell_price = df.iloc[1 + hold_days]['收盘']
            else:
                sell_price = df.iloc[-1]['收盘']
        
        return_rate = (sell_price - buy_price) / buy_price
        
        return {
            'buy_price': buy_price,
            'sell_price': sell_price,
            'return_rate': return_rate,
            'next_open': df.iloc[1]['开盘'],
            'next_high': df.iloc[1]['最高'],
            'next_low': df.iloc[1]['最低'],
            'next_close': df.iloc[1]['收盘'],
            'next_volume': df.iloc[1]['成交量'],
        }
    except Exception as e:
        return None

def backtest_lhb_strategy(
    start_date='20230101',
    end_date='20241201',
    top_n=5,          # 每天选净买入前N只
    hold_days=1,      # 持有天数
    filters=None       # 额外过滤条件
):

    回测龙虎榜净买入策略

    print(f"每日选股: 净买入前{top_n}名")
    print(f"持有天数: {hold_days}")
    print("=" * 60)
    
    # 1. 获取龙虎榜数据
    print("正在获取龙虎榜数据...")
    lhb_data = get_lhb_data(start_date, end_date)
    
    if lhb_data.empty:

    net_buy_df = calculate_net_buy(lhb_data)
    
    # 3. 筛选每日Top N
    selected = net_buy_df[net_buy_df['rank'] <= top_n].copy()
    
    # 4. 计算次日收益
    results = []

        stock_code = row['代码']
        trade_date = row['上榜日期']

            results.append({
                'stock_code': stock_code,
                'trade_date': trade_date,
                'net_buy': row['净买入额'],
                'rank': row['rank'],
                **ret
            })
        
        if len(results) % 50 == 0:
            print(f"进度: {len(results)}/{total}")
        
        time.sleep(0.3)
    
    # 5. 统计结果

    print(f"夏普比(日): {sharpe:.3f}")
```

## 三、纯净买入策略的历史表现（经验数据）

根据市场公开研究和量化社区的回测结论：

### 基础策略表现

| 指标 | 数值范围 | 说明 |
|------|----------|------|
| **胜率** | **48%~55%** | 牛市偏高，熊市偏低 |
| **平均日收益** | **0.3%~1.5%** | 波动大，不稳定 |
| **中位数收益** | **-0.2%~0.8%** | 被极端值拉偏 |

### 关键发现

> ⚠️ **单纯按净买入最大选股，胜率勉强接近55%，平均收益难以稳定超过2%**

原因：
1. 龙虎榜T日收盘后公布，T+1很多资金已经知道并抢筹
2. 净买入大不代表主力建仓，可能是对倒、出货的一部分
3. 大量跟风盘导致T+1高开低走

```python
def filter_by_institution(lhb_detail):
    """
    核心过滤器：只选机构专用席位净买入的股票
    机构席位 vs 游资席位 表现差异巨大
    """
    # 机构专用席位关键词
    institution_keywords = ['机构专用', '机构']
    
    # 筛选买入方含机构席位的记录
    inst_buy = lhb_detail[
        lhb_detail['买方营业部'].str.contains('|'.join(institution_keywords), na=False)
    ]
    
    # 筛选卖出方不含机构席位（机构只买不卖）
    inst_sell = lhb_detail[
        lhb_detail['卖方营业部'].str.contains('|'.join(institution_keywords), na=False)
    ]
    
    # 计算机构净买入
    inst_net = inst_buy.groupby('代码')['买入额'].sum() - \
               inst_sell.groupby('代码')['卖出额'].sum().reindex(inst_buy.groupby('代码').groups.keys(), fill_value=0)
    
    # 只选机构净买入 > 0 的
    selected = inst_net[inst_net > 0].index.tolist()
    
    return selected

# 【经验数据】
# 机构席位净买入的股票：
# 胜率: 55%~62%
# 平均T+1收益: 1.0%~2.5%
# 平均T+5收益: 2.0%~5.0%（持有更久效果更好）
```

```python
import talib

def calculate_technical_score(stock_code, trade_date):
    """
    技术面打分系统（0~100分）
    """
    # 获取近60日K线
    df = ak.stock_zh_a_hist(
        symbol=stock_code,
        period="daily",
        start_date=(pd.to_datetime(trade_date) - timedelta(days=90)).strftime('%Y%m%d'),
        end_date=trade_date,
        adjust="qfq"
    )
    
    if df is None or len(df) < 30:
        return 0
    
    close = df['收盘'].values
    high = df['最高'].values
    low = df['最低'].values
    volume = df['成交量'].values
    
    score = 0
    
    # ===== 因子1：趋势位置（30分）=====
    ma5 = talib.SMA(close, 5)
    ma10 = talib.SMA(close, 10)
    ma20 = talib.SMA(close, 20)
    ma60 = talib.SMA(close, 60)
    
    current_price = close[-1]
    
    # 价格在均线之上 → 趋势向上
    if current_price > ma5[-1]:
        score += 5
    if current_price > ma10[-1]:
        score += 5
    if current_price > ma20[-1]:
        score += 10
    if current_price > ma60[-1]:
        score += 10
    
    # ===== 因子2：量能配合（20分）=====
    vol_ma5 = talib.SMA(volume.astype(float), 5)
    vol_ma20 = talib.SMA(volume.astype(float), 20)
    
    # 近5日放量（量比 > 1.5）
    if vol_ma5[-1] > vol_ma20[-1] * 1.5:
        score += 10
    elif vol_ma5[-1] > vol_ma20[-1] * 1.2:
        score += 5
    
    # 当日成交量是20日均量的倍数
    vol_ratio = volume[-1] / vol_ma20[-1] if vol_ma20[-1] > 0 else 1
    if 2 < vol_ratio < 5:  # 适度放量最佳，过度放量可能是出货
        score += 10
    elif 1.5 < vol_ratio <= 2:
        score += 5
    
    # ===== 因子3：涨幅位置（20分）=====
    # 近20日累计涨幅
    gain_20d = (close[-1] / close[-20] - 1) if len(close) >= 20 else 0
    
    # 刚启动（涨幅 < 20%）比已经大涨（> 50%）好
    if 0 < gain_20d < 0.15:
        score += 20  # 刚启动，空间大
    elif 0.15 <= gain_20d < 0.30:
        score += 10  # 中段
    elif gain_20d >= 0.50:
        score -= 10  # 高位接盘风险大
    
    # ===== 因子4：K线形态（15分）=====
    # MACD金叉
    macd, signal, hist = talib.MACD(close, 12, 26, 9)
    if hist[-1] > 0 and hist[-2] <= 0:
        score += 10  # 刚金叉
    elif hist[-1] > hist[-2] > 0:
        score += 5   # MACD柱放大
    
    # RSI不超买
    rsi = talib.RSI(close, 14)
    if 40 < rsi[-1] < 70:
        score += 5   # 健康区间
    elif rsi[-1] > 80:
        score -= 5   # 超买
    
    # ===== 因子5：换手率（15分）=====
    # 需要流通股本数据来算换手率，这里用成交量近似
    # 当日换手适中（3%~10%为佳）
    # 此处简化处理
    score += 5  # 默认给分
    
    return max(0, min(100, score))


def enhanced_stock_selection(lhb_data, trade_date, top_n=3):
    """
    增强选股：龙虎榜净买入 + 技术面打分
    """
    candidates = lhb_data[lhb_data['上榜日期'] == trade_date].copy()
    
    # 只选净买入 > 0 的
    candidates = candidates[candidates['净买入额'] > 0]
    
    # 计算技术面得分
    candidates['tech_score'] = candidates['代码'].apply(
        lambda x: calculate_technical_score(x, trade_date)
    )
    
    # 综合评分 = 净买入排名分(40%) + 技术面分(60%)
    candidates['net_buy_score'] = candidates['净买入额'].rank(pct=True) * 100
    candidates['final_score'] = (
        candidates['net_buy_score'] * 0.4 + 
        candidates['tech_score'] * 0.6
    )
    
    # 过滤：技术面分 >= 50 才入选
    candidates = candidates[candidates['tech_score'] >= 50]
    
    # 取综合评分最高的N只
    selected = candidates.nlargest(top_n, 'final_score')
    
    return selected
```

```python
                continue
        except:
            continue
        
        # ====== 第三层：基础过滤 ======
        candidates = lhb_detail.copy()
        
        # 3.1 排除ST股
        candidates = candidates[~candidates['名称'].str.contains('ST|st', na=False)]
        
        # 3.2 排除上市不足60天的新股（容易暴跌）
        # candidates = filter_new_stocks(candidates, min_days=60)
        
        # 3.3 净买入额 > 5000万
        stock_net_buy = candidates.groupby('代码').agg({
            '名称': 'first',
            '买入额': 'sum',
            '卖出额': 'sum',
        }).reset_index()
        stock_net_buy['净买入额'] = stock_net_buy['买入额'] - stock_net_buy['卖出额']
        stock_net_buy = stock_net_buy[stock_net_buy['净买入额'] > 50000000]  # 5000万
        
        if stock_net_buy.empty:
            continue
        
        # ====== 第四层：席位质量过滤 ======
        qualified_stocks = []
        
        for _, row in stock_net_buy.iterrows():
            code = row['代码']
            stock_detail = candidates[candidates['代码'] == code]
            
            # 检查是否有机构席位买入
            has_institution = stock_detail['买方营业部'].str.contains(
                '机构专用', na=False
            ).any() if '买方营业部' in stock_detail.columns else False
            
            # 检查是否有知名游资买入
            famous_count = 0
            if '买方营业部' in stock_detail.columns:
                for dept in stock_detail['买方营业部'].dropna():
                    for name in FAMOUS_TRADERS.keys():
                        if name in str(dept):
                            famous_count += 1
            
            # 检查卖出方是否有机构（机构在卖 → 不好）
            inst_selling = stock_detail['卖方营业部'].str.contains(
                '机构专用', na=False
            ).any() if '卖方营业部' in stock_detail.columns else False
            
            # 席位评分
            seat_score = 0
            if has_institution and not inst_selling:
                seat_score += 40   # 机构买入且不卖出
            if has_institution and inst_selling:
                seat_score += 10   # 机构有分歧
            if famous_count >= 2:
                seat_score += 30   # 多游资合力
            elif famous_count == 1:
                seat_score += 15
            if inst_selling and not has_institution:
                seat_score -= 20   # 机构只卖不买
            
            if seat_score >= 15:  # 席位评分达标
                qualified_stocks.append({
                    'code': code,
                    'name': row['名称'],
                    'net_buy': row['净买入额'],
                    'seat_score': seat_score,
                    'has_institution': has_institution,
                    'famous_count': famous_count,
                })
        
        if not qualified_stocks:
            continue
        
        # ====== 第五层：技术面打分 ======
        for stock in qualified_stocks:
            tech_score = calculate_technical_score(stock['code'], date_str)
            stock['tech_score'] = tech_score
            
            # 获取近20日涨幅
            try:
                hist = ak.stock_zh_a_hist(
                    symbol=stock['code'],
                    period="daily",
                    start_date=(trade_date - timedelta(days=35)).strftime('%Y%m%d'),
                    end_date=date_str,
                    adjust="qfq"
                )
                if hist is not None and len(hist) >= 20:
                    gain_20d = hist['收盘'].iloc[-1] / hist['收盘'].iloc[-20] - 1
                    stock['gain_20d'] = gain_20d
                else:
                    stock['gain_20d'] = 0
            except:
                stock['gain_20d'] = 0
        
        # 过滤：技术面 >= 50 且 20日涨幅 < 30%
        qualified_stocks = [
            s for s in qualified_stocks 
            if s['tech_score'] >= 50 and s['gain_20d'] < 0.30
        ]
        
        if not qualified_stocks:
            continue
        
        # ====== 第六层：综合评分排序 ======
        for stock in qualified_stocks:
            # 净买入归一化（0~100）
            max_net = max(s['net_buy'] for s in qualified_stocks)
            min_net = min(s['net_buy'] for s in qualified_stocks)
            if max_net > min_net:
                net_score = (stock['net_buy'] - min_net) / (max_net - min_net) * 100
            else:
                net_score = 50
            
            # 综合评分
            stock['final_score'] = (
                net_score * 0.25 +           # 净买入额权重25%
                stock['seat_score'] * 0.35 + # 席位质量权重35%
                stock['tech_score'] * 0.30 + # 技术面权重30%
                (30 - stock['gain_20d'] * 100) * 0.10  # 低位加分权重10%
            )
        
        # 按综合评分排序，取前N只
        qualified_stocks.sort(key=lambda x: x['final_score'], reverse=True)
        selected = qualified_stocks[:top_n]
        
        # ====== 第七层：计算次日收益 ======
        for stock in selected:
            ret = get_next_day_return(stock['code'], date_str, hold_days)
            
            if ret:
                results.append({
                    'trade_date': date_str,
                    'code': stock['code'],
                    'name': stock['name'],
                    'net_buy': stock['net_buy'],
                    'seat_score': stock['seat_score'],
                    'tech_score': stock['tech_score'],
                    'final_score': stock['final_score'],
                    'has_institution': stock['has_institution'],
                    'famous_count': stock['famous_count'],
                    'gain_20d': stock['gain_20d'],
                    'position_ratio': position_ratio,
                    **ret
                })
        
        time.sleep(0.5)
    
    # ====== 汇总统计 ======
    return analyze_results(pd.DataFrame(results))
```

```python
def simulate_with_stoploss(stock_code, trade_date, 
                            stop_loss=-0.05,     # 止损 -5%
                            stop_profit=0.08,    # 止盈 +8%
                            max_hold_days=3):    # 最长持有3天
    """
    模拟带止盈止损的交易
    """
    try:
        df = ak.stock_zh_a_hist(
            symbol=stock_code,
            period="daily",
            start_date=trade_date,
            end_date=(pd.to_datetime(trade_date) + timedelta(days=max_hold_days + 10)).strftime('%Y%m%d'),
            adjust="qfq"
        )
        
        if df is None or len(df) < 2:
            return None
        
        buy_price = df.iloc[1]['开盘']  # T+1开盘买入
        
        for day_idx in range(1, len(df)):
            if day_idx > max_hold_days:
                # 超过最大持有天数，收盘强制卖出
                sell_price = df.iloc[day_idx]['收盘']
                return {
                    'buy_price': buy_price,
                    'sell_price': sell_price,
                    'return_rate': (sell_price - buy_price) / buy_price,
                    'hold_days': day_idx,
                    'exit_reason': '到期卖出'
                }
            
            day_open = df.iloc[day_idx]['开盘']
            day_high = df.iloc[day_idx]['最高']
            day_low = df.iloc[day_idx]['最低']
            day_close = df.iloc[day_idx]['收盘']
            
            # 计算盘中最高/最低收益率
            max_return = (day_high - buy_price) / buy_price
            min_return = (day_low - buy_price) / buy_price
            open_return = (day_open - buy_price) / buy_price
            
            # 开盘就触发止损（跳空低开）
            if day_idx > 1 and open_return <= stop_loss:
                sell_price = day_open
                return {
                    'buy_price': buy_price,
                    'sell_price': sell_price,
                    'return_rate': (sell_price - buy_price) / buy_price,
                    'hold_days': day_idx,
                    'exit_reason': '开盘止损'
                }
            
            # 开盘就触发止盈（跳空高开）
            if open_return >= stop_profit:
                sell_price = day_open
                return {
                    'buy_price': buy_price,
                    'sell_price': sell_price,
                    'return_rate': (sell_price - buy_price) / buy_price,
                    'hold_days': day_idx,
                    'exit_reason': '开盘止盈'
                }
            
            # 盘中触发止损
            if min_return <= stop_loss:
                sell_price = buy_price * (1 + stop_loss)
                return {
                    'buy_price': buy_price,
                    'sell_price': sell_price,
                    'return_rate': stop_loss,
                    'hold_days': day_idx,
                    'exit_reason': '盘中止损'
                }
            
            # 盘中触发止盈
            if max_return >= stop_profit:
                sell_price = buy_price * (1 + stop_profit)
                return {
                    'buy_price': buy_price,
                    'sell_price': sell_price,
                    'return_rate': stop_profit,
                    'hold_days': day_idx,
                    'exit_reason': '盘中止盈'
                }
        
        # 数据不足，按最后一天收盘价
        sell_price = df.iloc[-1]['收盘']
        return {
            'buy_price': buy_price,
            'sell_price': sell_price,
            'return_rate': (sell_price - buy_price) / buy_price,
            'hold_days': len(df) - 1,
            'exit_reason': '数据截止'
        }
        
    except Exception as e:
        print(f"止盈止损模拟失败 {stock_code}: {e}")
        return None
```

```python
        if df_min is None or df_min.empty:
            return 'open'  # 默认开盘买
        
        open_price = df_min.iloc[0]['开盘']
        
        # 分析日内走势模式
        # 模式1: 高开低走 → 等回调买入
        # 模式2: 低开高走 → 立即买入
        # 模式3: 冲高回落再拉升 → 回落时买入
        
        first_30min_high = df_min.iloc[:6]['最高'].max()  # 前30分钟最高
        first_30min_low = df_min.iloc[:6]['最低'].min()
        
        gap_up = (open_price / df_min.iloc[0]['开盘'] - 1)  # 跳空幅度
        
        if gap_up > 0.05:
            return 'wait_pullback'  # 高开太多，等回调
        elif gap_up > 0.02:
            return 'half_open_half_pullback'  # 半仓开盘买，半仓等回调
        else:
            return 'open'  # 直接开盘买
            
    except:
        return 'open'


def smart_entry_price(stock_code, trade_date, strategy='open'):
    """
    根据策略确定买入价
    """
    try:
        df = ak.stock_zh_a_hist(
            symbol=stock_code,
            period="daily",
            start_date=trade_date,
            end_date=(pd.to_datetime(trade_date) + timedelta(days=5)).strftime('%Y%m%d'),
            adjust="qfq"
        )
        
        if df is None or len(df) < 2:
            return None
        
        t1 = df.iloc[1]  # T+1日数据
        
        if strategy == 'open':
            return t1['开盘']
        
        elif strategy == 'wait_pullback':
            # 假设能在开盘价和最低价中间位置买入
            return (t1['开盘'] + t1['最低']) / 2
        
        elif strategy == 'half_open_half_pullback':
            # 平均价 = (开盘 + 中间价) / 2
            mid_price = (t1['开盘'] + t1['最低']) / 2
            return (t1['开盘'] + mid_price) / 2
        
        elif strategy == 'vwap':
            # 近似VWAP = (最高+最低+收盘) / 3
            return (t1['最高'] + t1['最低'] + t1['收盘']) / 3
        
        return t1['开盘']
    except:
        return None
```

```python
        """盘中止盈止损监控（每分钟运行）"""

            return

                row = df[df['代码'] == code]

                    continue
```
