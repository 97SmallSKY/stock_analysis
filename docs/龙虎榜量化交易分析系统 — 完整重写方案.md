### 3.2 完整 DDL

```sql
-- ============================================================
-- 0. 扩展
-- ============================================================
CREATE EXTENSION IF NOT EXISTS timescaledb;

-- ============================================================
-- 一、时序行情层 (TimescaleDB Hypertable)
-- ============================================================

-- 日K线（全市场，TimescaleDB 自动分区）
CREATE TABLE kline_daily (
    time            DATE            NOT NULL,
    stock_code      CHAR(6)         NOT NULL,
    open            NUMERIC(10,3)   NOT NULL,
    high            NUMERIC(10,3)   NOT NULL,
    low             NUMERIC(10,3)   NOT NULL,
    close           NUMERIC(10,3)   NOT NULL,
    volume          BIGINT          NOT NULL,
    turnover        NUMERIC(16,2),              -- 成交额（元）
    adjust_flag     CHAR(3)         NOT NULL DEFAULT 'raw'
                    CHECK (adjust_flag IN ('raw','qfq','hfq')),
    source          TEXT            NOT NULL DEFAULT 'tushare',
    UNIQUE (time, stock_code, adjust_flag)
);
SELECT create_hypertable('kline_daily', 'time');
CREATE INDEX idx_kline_code_time ON kline_daily (stock_code, time DESC);

-- 盘中快照（10秒级，TimescaleDB 自动分区）
CREATE TABLE market_snapshots (
    time            TIMESTAMPTZ     NOT NULL,
    stock_code      CHAR(6)         NOT NULL,
    price           NUMERIC(10,3)   NOT NULL,
    open            NUMERIC(10,3),
    high            NUMERIC(10,3),
    low             NUMERIC(10,3),
    volume          BIGINT,
    turnover        NUMERIC(16,2),
    bid1            NUMERIC(10,3),
    ask1            NUMERIC(10,3),
    source          TEXT            NOT NULL DEFAULT 'akshare',
    UNIQUE (time, stock_code)
);
SELECT create_hypertable('market_snapshots', 'time');
CREATE INDEX idx_snap_code_time ON market_snapshots (stock_code, time DESC);

-- 自动压缩（7天后压缩）
SELECT add_compression_policy('market_snapshots', INTERVAL '7 days');
-- 自动保留（90天后删除快照）
SELECT add_retention_policy('market_snapshots', INTERVAL '90 days');

-- ============================================================
-- 二、龙虎榜数据层 (普通 PostgreSQL 表)
-- ============================================================

CREATE TABLE lhb_stock_daily (
    id              SERIAL PRIMARY KEY,
    trade_date      DATE            NOT NULL,
    stock_code      CHAR(6)         NOT NULL,
    stock_name      TEXT            NOT NULL,
    close_price     NUMERIC(10,3)   NOT NULL,
    change_pct      NUMERIC(8,4),
```

```python
# backend/app/data_providers/base.py

from abc import ABC, abstractmethod
from enum import Enum
from typing import Optional, Any
import logging

logger = logging.getLogger(__name__)


class DataCategory(str, Enum):
    DAILY_KLINE      = "daily_kline"
    LHB_DETAIL       = "lhb_detail"
    LHB_BROKER       = "lhb_broker"
    INSTITUTION      = "institution"
    REALTIME_QUOTE   = "realtime_quote"
    MARKET_SENTIMENT = "market_sentiment"
    FINANCIAL_DATA   = "financial_data"


class ProviderStatus(str, Enum):
    OK       = "ok"
    DEGRADED = "degraded"     # 降级中（限频/部分失败）
    DOWN     = "down"         # 完全不可用


class BaseProvider(ABC):
    """数据源基类"""
    name: str = "base"
    supports: list[DataCategory] = []

    @abstractmethod
    async def fetch(self, category: DataCategory, **kwargs) -> Optional[Any]:
        raise NotImplementedError

    @abstractmethod
    async def health_check(self) -> ProviderStatus:
        raise NotImplementedError


# backend/app/data_providers/manager.py

class DataProviderManager:
    """
    数据源降级链管理器
    任意节点失败 → 自动降级下一节点 → 全部失败返回 None + 标记
    """

    # 优先级链配置
    CHAINS: dict[DataCategory, list[str]] = {
        DataCategory.DAILY_KLINE:      ["tushare", "akshare", "sina", "crawler"],
        DataCategory.LHB_DETAIL:       ["akshare", "sina", "crawler"],
        DataCategory.LHB_BROKER:       ["akshare", "crawler"],
        DataCategory.INSTITUTION:      ["sina", "crawler"],
        DataCategory.REALTIME_QUOTE:   ["akshare", "sina"],
        DataCategory.MARKET_SENTIMENT: ["akshare", "sina"],
        DataCategory.FINANCIAL_DATA:   ["tushare", "akshare"],
    }

    def __init__(self):
        self._providers: dict[str, BaseProvider] = {}
        self._failure_counts: dict[str, int] = {}       # 连续失败计数
        self._last_success: dict[str, float] = {}       # 上次成功时间戳

    def register(self, provider: BaseProvider):
        self._providers[provider.name] = provider
        self._failure_counts[provider.name] = 0

    async def fetch(
        self,
        category: DataCategory,
        **kwargs
    ) -> tuple[Optional[Any], str | None, list[str]]:
        """
        返回: (data, used_provider_name, failed_providers)
        全部失败: (None, None, [所有尝试过的 provider])
        """
        chain = self.CHAINS.get(category, [])
        failed = []

        for provider_name in chain:
            provider = self._providers.get(provider_name)
            if not provider:
                continue
            if category not in provider.supports:
                continue

            try:
                data = await provider.fetch(category, **kwargs)
                if data is not None:
                    self._failure_counts[provider_name] = 0
                    self._last_success[provider_name] = time.time()
                    logger.info(
                        f"[DataProvider] {category.value} ← {provider_name} 成功"
                    )
                    return data, provider_name, failed
                else:
                    failed.append(provider_name)
                    logger.warning(
                        f"[DataProvider] {category.value} ← {provider_name} 返回空"
                    )
            except Exception as e:
                self._failure_counts[provider_name] += 1
                failed.append(provider_name)
                logger.error(
                    f"[DataProvider] {category.value} ← {provider_name} 异常: {e}"
                )
                continue

        # 全部失败
        logger.error(
            f"[DataProvider] {category.value} 全部失败: {failed}"
        )
        return None, None, failed
```

```python
        # 3. 对每只股票调用 scorer 评分

        # 6. 返回 top_n 信号

        """
        等额分配: cash / max_positions * position_mult
        向下取整到 100 股
        """
        ...

    async def check_sell(self, position, current_price, hold_days, params):
        """
        止盈: +3%(1/3) → +5%(1/3) → +8%(清仓)
        止损: -5% 清仓
        到期: hold_days >= max_hold_days 清仓
        """
        buy_price = position['buy_price']
        pnl_pct = (current_price - buy_price) / buy_price

        if pnl_pct <= params['stop_loss_pct']:
            return SellDecision(True, 'stop_loss', 1.0, f'止损 {pnl_pct:.2%}')

        if pnl_pct >= params['sell3_pct'] and not position['sell3_done']:
            return SellDecision(True, 'sell3', 1.0, f'三档止盈 {pnl_pct:.2%}')

        if pnl_pct >= params['sell2_pct'] and not position['sell2_done']:
            return SellDecision(True, 'sell2', params['sell2_ratio'], f'二档止盈')

        if pnl_pct >= params['sell1_pct'] and not position['sell1_done']:
            return SellDecision(True, 'sell1', params['sell1_ratio'], f'一档止盈')

        if hold_days >= params['max_hold_days']:
            return SellDecision(True, 'expire', 1.0, f'到期强平 {hold_days}天')

        return SellDecision(False, '', 0, '')

    async def filter_t1_open(self, signal, open_price, prev_close, params):
        """
        T+1 开盘价过滤规则:
        涨停(>=10%): 放弃
        >7%: 放弃
        2%~7%: 半仓
        -1%~2%: 全仓（最佳）
        -3%~-1%: 半仓
        <-3%: 放弃
        """
        pct = (open_price - prev_close) / prev_close
        if pct >= 0.10:   return False, 0
        if pct > 0.07:    return False, 0
        if pct > 0.02:    return True, 0.5
        if pct >= -0.01:  return True, 1.0
        if pct > -0.03:   return True, 0.5
        return False, 0
```

```python
class RiskService:

    # ──────────────────────────────────────
    # AHS 评分（0~100）
    # ──────────────────────────────────────

    async def calc_ahs(self, robot_id: int) -> dict:
        """
        AHS = 50（基础）
            + 近10笔胜率贡献（±25）
            + 近10笔均盈亏贡献（±15）
            + 连亏惩罚（最多-20，连亏≥3次）
            + 今日盈亏修正（±10）
        """
        # 获取近10笔已平仓交易
        stmt = (
            select(RobotTrade)
            .where(
                RobotTrade.robot_id == robot_id,
                RobotTrade.status == 'closed'
            )
            .order_by(RobotTrade.sell_date.desc())
            .limit(10)
        )
        result = await self.db.execute(stmt)
        recent = result.scalars().all()

        if not recent:
            return {
                'ahs_score': 50,
                'win_rate': 0,
                'avg_pnl': 0,
                'consecutive_loss': 0,
                'position_mult': 1.0,
                'details': {'reason': '无历史交易'}
            }

        # 胜率贡献 (±25)
        wins = sum(1 for t in recent if t.profit_pct and t.profit_pct > 0)
        win_rate = wins / len(recent)
        win_contribution = round((win_rate - 0.5) * 50, 2)   # 50%=0, 100%=+25
        win_contribution = max(-25, min(25, win_contribution))

        # 均盈亏贡献 (±15)
        avg_pnl = sum(t.profit_pct or 0 for t in recent) / len(recent)
        pnl_contribution = round(avg_pnl * 300, 2)           # 5%=+15
        pnl_contribution = max(-15, min(15, pnl_contribution))

        # 连亏惩罚 (最多-20)
        consecutive_loss = 0
        for t in recent:
            if t.profit_pct and t.profit_pct < 0:
                consecutive_loss += 1
            else:
                break
        loss_penalty = 0
        if consecutive_loss >= 5:
            loss_penalty = -20
        elif consecutive_loss >= 3:
            loss_penalty = -15
        elif consecutive_loss >= 2:
            loss_penalty = -8

        # 今日盈亏修正 (±10)
        today_trades = [t for t in recent if t.sell_date == date.today()]
        today_pnl = sum(t.profit_pct or 0 for t in today_trades) if today_trades else 0
        today_mod = round(today_pnl * 200, 2)
        today_mod = max(-10, min(10, today_mod))

        # 汇总
        ahs = 50 + win_contribution + pnl_contribution + loss_penalty + today_mod
        ahs = max(0, min(100, round(ahs)))

        # 仓位乘数
        if ahs >= 75:
            position_mult = 1.2
        elif ahs >= 60:
            position_mult = 1.0
        elif ahs >= 50:
            position_mult = 0.75
        elif ahs >= 30:
            position_mult = 0.5
        else:
            position_mult = 0.1

        return {
            'ahs_score': ahs,
            'win_rate': round(win_rate, 4),
            'avg_pnl': round(avg_pnl, 6),
            'consecutive_loss': consecutive_loss,
            'position_mult': position_mult,
            'details': {
                'win_contribution': win_contribution,
                'pnl_contribution': pnl_contribution,
                'loss_penalty': loss_penalty,
                'today_mod': today_mod,
                'trade_count': len(recent),
            }
        }

        result = await self.db.execute(stmt)
        closed_trades = result.scalars().all()

        today = date.today()

        # ① 单日亏损 > 3%
        today_trades = [t for t in closed_trades if t.sell_date == today]
        if today_trades:
            today_pnl = sum(t.profit or 0 for t in today_trades)
            today_loss_pct = today_pnl / float(robot.initial_capital)
            details['today_loss_pct'] = round(today_loss_pct, 6)
            if today_loss_pct < -0.03:
                triggers.append('single_day_loss')

        # ② 连续亏损 >= 3 笔
        consecutive_loss = 0
        for t in closed_trades:
            if t.profit_pct and t.profit_pct < 0:
                consecutive_loss += 1
            else:
                break
        details['consecutive_loss'] = consecutive_loss
        if consecutive_loss >= 3:
            triggers.append('consecutive_losses')

        # ③ 周亏损 > 5%

        week_trades = [
            t for t in closed_trades
            if t.sell_date and t.sell_date >= week_start
        ]
        if week_trades:
            week_pnl = sum(t.profit or 0 for t in week_trades)
            week_loss_pct = week_pnl / float(robot.initial_capital)
            details['weekly_loss_pct'] = round(week_loss_pct, 6)
            if week_loss_pct < -0.05:
                triggers.append('weekly_loss')

        month_start = today.replace(day=1)
        month_trades = [
            t for t in closed_trades
            if t.sell_date and t.sell_date >= month_start
        ]
        if month_trades:
            month_pnl = sum(t.profit or 0 for t in month_trades)
            month_loss_pct = month_pnl / float(robot.initial_capital)
            details['monthly_loss_pct'] = round(month_loss_pct, 6)
            if month_loss_pct < -0.08:
                triggers.append('monthly_loss')

            select(DailyNav)
```

```python
            float(robot.commission_min)
        )
        slippage = amount * float(robot.slippage_buy)

        return {
            'amount': round(amount, 2),
            'commission': round(commission, 4),
            'slippage': round(slippage, 4),
            'total_cost': round(amount + commission + slippage, 4),
        }

    def calc_sell_cost(self, robot: Robot, price: float, shares: int,
                       stock_code: str) -> dict:
        """
        卖出成本:
        - 佣金: max(amount * commission_rate, commission_min)
        - 印花税: amount * stamp_tax_rate（仅卖出）
        - 过户费: amount * transfer_rate（仅沪市 6 开头）
        - 滑点: amount * slippage_sell
        """
        amount = price * shares
        commission = max(
            amount * float(robot.commission_rate),
            float(robot.commission_min)
        )
        stamp_tax = amount * float(robot.stamp_tax_rate)

        # 过户费仅沪市
        transfer = 0.0
        if stock_code.startswith('6'):
            transfer = amount * float(robot.transfer_rate)

        slippage = amount * float(robot.slippage_sell)

        total_cost = commission + stamp_tax + transfer + slippage

        return {
            'amount': round(amount, 2),
            'commission': round(commission, 4),
            'stamp_tax': round(stamp_tax, 4),
            'transfer': round(transfer, 4),
            'slippage': round(slippage, 4),
            'total_cost': round(total_cost, 4),
            'net_income': round(amount - total_cost, 4),
        }

    # ──────────────────────────────────────
    # Phase A: 结算 + 选股（每日 18:50）
    # ──────────────────────────────────────

    async def phase_a(self, robot_id: int, signal_date: date):
        """
        Phase A 完整流程:
        1. 结算过期持仓
        2. AHS 计算
        3. 熔断检查
        4. 策略选股
        5. 写入 pending 交易
        6. 审计日志
        """
        async with self.db.begin():
            robot = await self.db.get(Robot, robot_id)
            if not robot or robot.status != 'running':
                return {'skipped': True, 'reason': f'机器人状态: {robot.status}'}

            # 1. 结算过期持仓
            settled = await self._settle_expired(robot, signal_date)

            # 2. AHS 计算
            ahs = await self.risk.calc_ahs(robot_id)
            await self._save_ahs(robot_id, signal_date, ahs)

            # 3. 熔断检查
            triggers, cb_details = await self.risk.check_circuit_breaker(robot_id)
            if 'max_drawdown' in triggers:
                await self._force_liquidate(robot, signal_date, '最大回撤超15%')
                return {
                    'action': 'liquidated',
                    'triggers': triggers,
                    'details': cb_details
                }
            if triggers:
                await self._log_audit(robot_id, None, 'circuit_breaker', {
                    'triggers': triggers, 'details': cb_details
                })
                return {
                    'action': 'circuit_breaker',
                    'triggers': triggers,
                    'details': cb_details
                }

            # 4. 策略选股
            strategy = get_strategy(robot.strategy_name)
            params = await self._load_strategy_params(robot.strategy_name)

            # 检查当前持仓数
            open_count = await self._count_open_positions(robot_id)
            available_slots = robot.max_positions - open_count
            if available_slots <= 0:
                return {'action': 'full_position', 'open_count': open_count}

            signals = await strategy.select_stocks(signal_date, robot.__dict__, params)
            signals = signals[:available_slots]

            # 5. 写入 pending 交易
            created_trades = []
            for sig in signals:
                buy_amount, buy_shares = await strategy.calc_buy_amount(
                    sig, float(robot.current_cash),
                    ahs['ahs_score'], ahs['position_mult'], params
                )
                if buy_shares <= 0:
                    continue

                trade = RobotTrade(
                    robot_id=robot_id,
                    signal_date=signal_date,
                    stock_code=sig.stock_code,
                    stock_name=sig.stock_name,
                    signal_score=sig.signal_score,
                    tech_score=sig.tech_score,
                    fund_score=sig.fund_score,
                    hist_score=sig.hist_score,
                    seat_tags=sig.seat_tags,
                    buy_amount=buy_amount,
                    buy_shares=buy_shares,
                    list_reason=sig.reason,
                    llm_reason=getattr(sig, 'llm_reason', None),
                    status='pending',
                    lianban_count=getattr(sig, 'lianban_count', 0),
                )
                self.db.add(trade)
                created_trades.append(trade)

            # 6. 审计日志
            await self._log_audit(robot_id, None, 'phase_a', {
                'signal_date': str(signal_date),
                'ahs': ahs,
                'signals_count': len(signals),
                'created_count': len(created_trades),
                'settled_count': len(settled),
            })

            return {
                'action': 'phase_a_done',
                'ahs': ahs,
                'created': len(created_trades),
                'settled': len(settled),
            }

    # ──────────────────────────────────────
    # Phase B: T+1 开盘买入（次日 9:35）
    # ──────────────────────────────────────

    async def phase_b(self, robot_id: int, buy_date: date):
        """
        Phase B 完整流程:
        1. 获取 pending 交易
        2. 获取实时开盘价
        3. T+1 规则过滤
        4. 执行买入（写 trade + cash_ledger）
        5. 审计日志
        """
        async with self.db.begin():
            robot = await self.db.get(Robot, robot_id)
            if not robot or robot.status != 'running':
                return {'skipped': True}

            # 1. 获取 pending 交易
            stmt = (
                select(RobotTrade)
                .where(
                    RobotTrade.robot_id == robot_id,
                    RobotTrade.status == 'pending'
                )
            )
            result = await self.db.execute(stmt)
            pending = result.scalars().all()

            strategy = get_strategy(robot.strategy_name)
            params = await self._load_strategy_params(robot.strategy_name)

            bought = []
            abandoned = []

            for trade in pending:
                # 2. 获取实时开盘价
                from app.data_providers.manager import data_manager
                quote, _, _ = await data_manager.fetch(
                    DataCategory.REALTIME_QUOTE,
                    stock_code=trade.stock_code
                )
                if not quote:
                    trade.status = 'abandoned'
                    abandoned.append(trade)
                    continue

                open_price = quote['open']
                prev_close = quote['prev_close']

                # 3. T+1 规则过滤
                should_buy, ratio = await strategy.filter_t1_open(
                    None, open_price, prev_close, params
                )
                if not should_buy:
                    trade.status = 'abandoned'
                    trade.list_reason = (trade.list_reason or '') + \
                        f' | T+1过滤: open={open_price}, pct={((open_price-prev_close)/prev_close):.2%}'
                    abandoned.append(trade)
                    continue

                # 调整买入数量（按 ratio）
                actual_shares = int(trade.buy_shares * ratio // 100) * 100
                if actual_shares <= 0:
                    trade.status = 'abandoned'
                    abandoned.append(trade)
                    continue

                # 4. 执行买入
                buy_cost = self.calc_buy_cost(robot, open_price, actual_shares)

                if buy_cost['total_cost'] > float(robot.current_cash):
                    trade.status = 'abandoned'
                    trade.list_reason = (trade.list_reason or '') + ' | 资金不足'
                    abandoned.append(trade)
                    continue

                # 更新交易记录
                trade.buy_date = buy_date
                trade.buy_price = open_price
                trade.buy_shares = actual_shares
                trade.buy_amount = buy_cost['amount']
                trade.buy_commission = buy_cost['commission']
                trade.buy_slippage = buy_cost['slippage']
                trade.buy_total_cost = buy_cost['total_cost']
                trade.status = 'open'

                # 扣减现金
                robot.current_cash = Decimal(str(
                    float(robot.current_cash) - buy_cost['total_cost']
                ))

                # 写现金流水
                ledger = CashLedger(
                    robot_id=robot_id,
                    trade_date=buy_date,
                    event_type='buy_cost',
                    amount=Decimal(str(-buy_cost['total_cost'])),
                    balance=robot.current_cash,
                    trade_id=trade.id,
                    note=f"买入 {trade.stock_code} {actual_shares}股 @{open_price}"
                )
                self.db.add(ledger)

                # 写事件
                event = TradeEvent(
                    trade_id=trade.id,
                    robot_id=robot_id,
                    event_type='phase_b',
                    price=open_price,
                    shares=actual_shares,
                    amount=buy_cost['total_cost'],
                    description=f"T+1买入 ratio={ratio}"
                )
                self.db.add(event)

                bought.append(trade)

            # 5. 审计日志
            await self._log_audit(robot_id, None, 'phase_b', {
                'buy_date': str(buy_date),
                'bought_count': len(bought),
                'abandoned_count': len(abandoned),
                'remaining_cash': str(robot.current_cash),
                'bought_codes': [t.stock_code for t in bought],
                'abandoned_codes': [t.stock_code for t in abandoned],
            })

            return {
                'action': 'phase_b_done',
                'bought': len(bought),
                'abandoned': len(abandoned),
                'remaining_cash': float(robot.current_cash),
            }

    # ──────────────────────────────────────
    # 卖出执行（盘中监控/到期强平调用）
    # ──────────────────────────────────────

    async def execute_sell(
        self,
        robot: Robot,
        trade: RobotTrade,
        sell_price: float,
        sell_type: str,
        sell_ratio: float,
        sell_date: date,
        reason: str = ''
    ) -> dict:
        """
        统一卖出执行入口
        sell_ratio: 卖出比例 (0~1)，1.0=清仓
        """
        sell_shares = int(trade.buy_shares * sell_ratio // 100) * 100
        if sell_shares <= 0:
            sell_shares = trade.buy_shares  # 剩余全卖

        # 计算卖出费用
        sell_cost = self.calc_sell_cost(
            robot, sell_price, sell_shares, trade.stock_code
        )

        # 更新交易记录
        trade.sell_price = sell_price
        trade.sell_shares = sell_shares
        trade.sell_amount = sell_cost['amount']
        trade.sell_commission = sell_cost['commission']
        trade.sell_stamp_tax = sell_cost['stamp_tax']
        trade.sell_transfer = sell_cost['transfer']
        trade.sell_slippage = sell_cost['slippage']
        trade.sell_total_cost = sell_cost['total_cost']
        trade.sell_net_income = sell_cost['net_income']
        trade.sell_date = sell_date
        trade.sell_type = sell_type

        # 计算盈亏
        trade.profit = Decimal(str(
            sell_cost['net_income'] - float(trade.buy_total_cost)
        ))
        trade.profit_pct = float(trade.profit) / float(trade.buy_total_cost)

        # 计算持仓天数
        if trade.buy_date:
            trade.hold_days = (sell_date - trade.buy_date).days

        # 更新卖出标记
        if sell_type == 'sell1':
            trade.sell1_done = True
        elif sell_type == 'sell2':
            trade.sell2_done = True
        elif sell_type == 'sell3':
            trade.sell3_done = True
        elif sell_type == 'stop_loss':
            trade.stop_loss_done = True

        # 判断是否完全平仓
        if sell_ratio >= 1.0 or sell_type in ('stop_loss', 'sell3', 'expire', 'manual'):
            trade.status = 'closed'
        else:
            trade.status = 'partial'

        # 现金入账
        robot.current_cash = Decimal(str(
            float(robot.current_cash) + sell_cost['net_income']
        ))

        # 写现金流水
        ledger = CashLedger(
            robot_id=robot.id,
            trade_date=sell_date,
            event_type='sell_income',
            amount=Decimal(str(sell_cost['net_income'])),
            balance=robot.current_cash,
            trade_id=trade.id,
            note=f"卖出 {trade.stock_code} {sell_shares}股 @{sell_price} [{sell_type}]"
        )
        self.db.add(ledger)

        # 写交易事件
        event = TradeEvent(
            trade_id=trade.id,
            robot_id=robot.id,
            event_type=sell_type,
            price=sell_price,
            shares=sell_shares,
            amount=sell_cost['net_income'],
            description=reason
        )
        self.db.add(event)

        # 审计日志
        await self._log_audit(robot.id, trade.id, f'sell_{sell_type}', {
            'sell_price': sell_price,
            'sell_shares': sell_shares,
            'profit': str(trade.profit),
            'profit_pct': f'{trade.profit_pct:.4%}',
            'reason': reason,
            'cost_detail': sell_cost,
        })

        return {
            'trade_id': trade.id,
            'sell_type': sell_type,
            'sell_price': sell_price,
            'sell_shares': sell_shares,
            'profit': float(trade.profit),
            'profit_pct': trade.profit_pct,
            'net_income': sell_cost['net_income'],
            'status': trade.status,
        }

    async def _settle_expired(self, robot: Robot, current_date: date) -> list[dict]:
        """结算超期持仓"""
        stmt = (
            select(RobotTrade)
            .where(
                RobotTrade.robot_id == robot.id,
                RobotTrade.status.in_(['open', 'partial']),
            )
        )
        result = await self.db.execute(stmt)
        open_trades = result.scalars().all()

        settled = []
        for trade in open_trades:
            if not trade.buy_date:
                continue
            hold_days = (current_date - trade.buy_date).days
            if hold_days >= robot.max_hold_days:
                # 获取收盘价
                from app.data_providers.manager import data_manager
                kline, _, _ = await data_manager.fetch(
                    DataCategory.DAILY_KLINE,
                    stock_code=trade.stock_code,
                    trade_date=current_date
                )
                if kline is not None and len(kline) > 0:
                    close_price = float(kline.iloc[-1]['close'])
                    result = await self.execute_sell(
                        robot, trade, close_price, 'expire', 1.0,
                        current_date, f'到期强平 {hold_days}天'
                    )
                    settled.append(result)
        return settled
```

```python
# backend/app/services/monitor_service.py

import asyncio
from datetime import date, datetime
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.robot import Robot, RobotTrade
from app.services.trade_service import TradeService
from app.services.notification import NotificationService
from app.strategies.registry import get_strategy
from app.data_providers.base import DataCategory
from app.core.redis_client import redis_client
import logging

logger = logging.getLogger(__name__)


class MonitorService:
    """盘中实时监控 — 每 10 秒轮询"""

    def __init__(
        self,
        db: AsyncSession,
        trade_service: TradeService,
        notification: NotificationService,
    ):
        self.db = db
        self.trade = trade_service
        self.notify = notification
        self._running = False

    async def start(self):
        """启动监控循环"""
        self._running = True
        logger.info("[Monitor] 盘中监控启动")

        while self._running:
            now = datetime.now()

            # 仅交易时段运行 9:30~15:00
            if not self._is_trading_hours(now):
                await asyncio.sleep(10)
                continue

            try:
                await self._monitor_cycle()
            except Exception as e:
                logger.error(f"[Monitor] 监控循环异常: {e}", exc_info=True)

            await asyncio.sleep(10)  # 10秒间隔

    async def stop(self):
        self._running = False
        logger.info("[Monitor] 盘中监控停止")

    async def _monitor_cycle(self):
        """单次监控循环：遍历所有运行中机器人的持仓"""

        # 1. 获取所有运行中的机器人
        stmt = select(Robot).where(Robot.status == 'running')
        result = await self.db.execute(stmt)
        robots = result.scalars().all()

        # 2. 收集所有需要监控的股票代码（去重，批量获取行情）
        all_codes = set()
        robot_positions = {}

        for robot in robots:
            pos_stmt = (
                select(RobotTrade)
                .where(
                    RobotTrade.robot_id == robot.id,
                    RobotTrade.status.in_(['open', 'partial']),
                )
            )
            pos_result = await self.db.execute(pos_stmt)
            positions = pos_result.scalars().all()

            if positions:
                robot_positions[robot.id] = {
                    'robot': robot,
                    'positions': positions,
                }
                for pos in positions:
                    all_codes.add(pos.stock_code)

        if not all_codes:
            return

        # 3. 批量获取实时行情（Redis 缓存 + 数据源降级链）
        quotes = await self._batch_fetch_quotes(list(all_codes))

        # 4. 逐机器人、逐持仓检查
        for robot_id, data in robot_positions.items():
            robot = data['robot']
            strategy = get_strategy(robot.strategy_name)
            params = await self.trade._load_strategy_params(robot.strategy_name)

            for position in data['positions']:
                quote = quotes.get(position.stock_code)
                if not quote:
                    continue

                current_price = quote['price']
                hold_days = 0
                if position.buy_date:
                    hold_days = (date.today() - position.buy_date).days

                # 调用策略判断
                decision = await strategy.check_sell(
                    position.__dict__, current_price, hold_days, params
                )

                if decision.should_sell:
                    async with self.db.begin():
                        sell_result = await self.trade.execute_sell(
                            robot, position,
                            current_price, decision.sell_type,
                            decision.sell_ratio, date.today(),
                            decision.reason
                        )

                    # 飞书通知
                    await self.notify.send_trade_alert(robot, position, sell_result)

                    # WebSocket 推送
                    await self._push_ws_event(robot_id, position, sell_result)

                # 更新 Redis 缓存（实时 PnL）
                if position.buy_price:
                    pnl_pct = (current_price - float(position.buy_price)) / float(position.buy_price)
                    await redis_client.hset(
                        f"pnl:{robot_id}",
                        position.stock_code,
                        f"{pnl_pct:.6f}"
                    )

    async def _batch_fetch_quotes(self, codes: list[str]) -> dict:
        """批量获取行情，优先 Redis 缓存"""
        quotes = {}
        uncached = []

        # 先查 Redis
        for code in codes:
            cached = await redis_client.hgetall(f"quote:{code}")
            if cached and 'price' in cached:
                quotes[code] = {
                    'price': float(cached['price']),
                    'open': float(cached.get('open', 0)),
                    'high': float(cached.get('high', 0)),
                    'low': float(cached.get('low', 0)),
                    'prev_close': float(cached.get('prev_close', 0)),
                    'volume': int(cached.get('volume', 0)),
                    'source': 'redis_cache',
                }
            else:
                uncached.append(code)

        # 未缓存的从数据源获取
        if uncached:
            from app.data_providers.manager import data_manager

            for code in uncached:
                data, provider, failed = await data_manager.fetch(
                    DataCategory.REALTIME_QUOTE,
                    stock_code=code
                )
                if data:
                    quotes[code] = data
                    # 写入 Redis 缓存（10秒过期）
                    await redis_client.hset(f"quote:{code}", mapping={
                        'price': str(data['price']),
                        'open': str(data.get('open', 0)),
                        'high': str(data.get('high', 0)),
                        'low': str(data.get('low', 0)),
                        'prev_close': str(data.get('prev_close', 0)),
                        'volume': str(data.get('volume', 0)),
                        'source': provider or 'unknown',
                    })
                    await redis_client.expire(f"quote:{code}", 10)
                else:
                    logger.warning(
                        f"[Monitor] {code} 行情获取失败, 已尝试: {failed}"
                    )

        return quotes

    async def _push_ws_event(self, robot_id: int, position, sell_result: dict):
        """通过 Redis pub/sub 推送 WebSocket 事件"""
        import json
        event = {
            'type': 'order_update',
            'robot_id': robot_id,
            'stock_code': position.stock_code,
            'stock_name': position.stock_name,
            'sell_type': sell_result['sell_type'],
            'sell_price': sell_result['sell_price'],
            'profit_pct': sell_result['profit_pct'],
            'status': sell_result['status'],
            'timestamp': datetime.now().isoformat(),
        }
        await redis_client.publish(
            f"ws:robot:{robot_id}", json.dumps(event, ensure_ascii=False)
        )

    def _is_trading_hours(self, now: datetime) -> bool:
        """判断当前是否在交易时段"""
        t = now.time()
        morning = time(9, 25) <= t <= time(11, 35)
        afternoon = time(12, 55) <= t <= time(15, 5)
        return (morning or afternoon) and now.weekday() < 5
```

### 4.7 淘汰评分引擎

```python
# backend/app/services/elimination_service.py

from datetime import date
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.robot import Robot, RobotTrade, DailyNav
import numpy as np
import logging

logger = logging.getLogger(__name__)


class EliminationService:
    """
    淘汰评分引擎 — 每周五执行
    
    指标         权重    计算
    ──────────────────────────────────────
    累计收益率    25%    (total_nav - initial) / initial
    最大回撤      20%    daily_nav 曲线历史最大回撤
    夏普比率      20%    日收益率序列, 无风险 2.5%/年
    胜率          10%    win / total
    盈亏比        10%    avg_win / abs(avg_loss)
    交易成本占比  10%    总费用 / 总成交额
    样本数量       5%    min(total / 20, 1.0)
    """

    WEIGHTS = {
        'return':      0.25,
        'drawdown':    0.20,
        'sharpe':      0.20,
        'win_rate':    0.10,
        'profit_loss': 0.10,
        'cost_ratio':  0.10,
        'sample_size': 0.05,
    }

    def __init__(self, db: AsyncSession):
        self.db = db

    async def calc_score(self, robot_id: int) -> dict:
        """计算单个机器人的淘汰评分（0~100，越高越好）"""

        robot = await self.db.get(Robot, robot_id)
        if not robot:
            return {'score': 0, 'error': 'robot_not_found'}

        # 获取全部已平仓交易
        stmt = (
            select(RobotTrade)
            .where(
                RobotTrade.robot_id == robot_id,
                RobotTrade.status == 'closed'
            )
            .order_by(RobotTrade.sell_date)
        )
        result = await self.db.execute(stmt)
        trades = result.scalars().all()

        # 获取净值序列
        nav_stmt = (
            select(DailyNav)
            .where(DailyNav.robot_id == robot_id)
            .order_by(DailyNav.nav_date)
        )
        nav_result = await self.db.execute(nav_stmt)
        navs = nav_result.scalars().all()

        scores = {}

        # ① 累计收益率 (25%)
        if navs:
            total_nav = float(navs[-1].total_nav)
            initial = float(robot.initial_capital)
            cum_return = (total_nav - initial) / initial
            # 归一化: -20%=0, 0%=50, +20%=100
            scores['return'] = self._normalize(cum_return, -0.20, 0.20)
        else:
            scores['return'] = 50

        # ② 最大回撤 (20%)
        if navs:
            nav_values = [float(n.total_nav) for n in navs]
            max_dd = self._calc_max_drawdown(nav_values)
            # 归一化: -30%=0, 0%=100
            scores['drawdown'] = self._normalize(-abs(max_dd), -0.30, 0)
        else:
            scores['drawdown'] = 50

        # ③ 夏普比率 (20%)
        if len(navs) >= 5:
            nav_values = [float(n.total_nav) for n in navs]
            daily_returns = np.diff(nav_values) / nav_values[:-1]
            rf_daily = 0.025 / 252  # 无风险利率
            excess = daily_returns - rf_daily
            if np.std(excess) > 0:
                sharpe = np.mean(excess) / np.std(excess) * np.sqrt(252)
            else:
                sharpe = 0
            # 归一化: -2=0, 0=50, +3=100
            scores['sharpe'] = self._normalize(sharpe, -2, 3)
        else:
            scores['sharpe'] = 50

        # ④ 胜率 (10%)
        if trades:
            wins = sum(1 for t in trades if t.profit_pct and t.profit_pct > 0)
            win_rate = wins / len(trades)
            # 归一化: 20%=0, 50%=50, 80%=100
            scores['win_rate'] = self._normalize(win_rate, 0.20, 0.80)
        else:
            scores['win_rate'] = 50

        # ⑤ 盈亏比 (10%)
        if trades:
            win_amounts = [float(t.profit_pct) for t in trades if t.profit_pct and t.profit_pct > 0]
            loss_amounts = [abs(float(t.profit_pct)) for t in trades if t.profit_pct and t.profit_pct < 0]

            avg_win = np.mean(win_amounts) if win_amounts else 0
            avg_loss = np.mean(loss_amounts) if loss_amounts else 1

            if avg_loss > 0:
                profit_loss_ratio = avg_win / avg_loss
            else:
                profit_loss_ratio = 3.0  # 无亏损

            # 归一化: 0.5=0, 1.0=50, 3.0=100
            scores['profit_loss'] = self._normalize(profit_loss_ratio, 0.5, 3.0)
        else:
            scores['profit_loss'] = 50

        # ⑥ 交易成本占比 (10%)
        if trades:
            total_cost = sum(
                float(t.buy_commission or 0) + float(t.buy_slippage or 0) +
                float(t.sell_commission or 0) + float(t.sell_stamp_tax or 0) +
                float(t.sell_transfer or 0) + float(t.sell_slippage or 0)
                for t in trades
            )
            total_volume = sum(
                float(t.buy_amount or 0) + float(t.sell_amount or 0)
                for t in trades
            )
            cost_ratio = total_cost / total_volume if total_volume > 0 else 0
            # 归一化: 1%=0, 0.3%=50, 0.1%=100（越低越好）
            scores['cost_ratio'] = self._normalize(-cost_ratio, -0.01, -0.001)
        else:
            scores['cost_ratio'] = 50

        # ⑦ 样本数量 (5%)
        sample_factor = min(len(trades) / 20, 1.0)
        scores['sample_size'] = sample_factor * 100

        # 加权汇总
        total = sum(
            scores[k] * self.WEIGHTS[k]
            for k in self.WEIGHTS
        )
        total = round(max(0, min(100, total)), 2)

        return {
            'robot_id': robot_id,
            'robot_name': robot.name,
            'elimination_score': total,
            'scores': scores,
            'trade_count': len(trades),
            'nav_days': len(navs),
        }

    async def run_weekly_elimination(self) -> dict:
        """
        每周淘汰执行:
        1. 计算所有运行中机器人评分
        2. 评分最低者标记为 eliminated
        3. 强制清仓
        4. 记录审计日志
        """
        stmt = select(Robot).where(Robot.status == 'running')
        result = await self.db.execute(stmt)
        robots = result.scalars().all()

        if len(robots) <= 1:
            return {'action': 'skip', 'reason': '运行中机器人不足2个'}

        # 计算所有评分
        rankings = []
        for robot in robots:
            score = await self.calc_score(robot.id)
            rankings.append(score)

        rankings.sort(key=lambda x: x['elimination_score'])

        # 淘汰最低分
        eliminated = rankings[0]
        robot = await self.db.get(Robot, eliminated['robot_id'])

        if robot:
            # 强制清仓
            from app.services.trade_service import TradeService
            from app.services.risk_service import RiskService
            risk = RiskService(self.db)
            trade_svc = TradeService(self.db, risk)
            await trade_svc._force_liquidate(
                robot, date.today(),
                f"周淘汰: 评分 {eliminated['elimination_score']}"
            )

            robot.status = 'eliminated'
            robot.elimination_score = eliminated['elimination_score']
            robot.ended_at = datetime.now()

            # 审计日志
            from app.models.event import AuditLog
            log = AuditLog(
                robot_id=robot.id,
                event_type='elimination',
                event_data={
                    'score': eliminated['elimination_score'],
                    'scores': eliminated['scores'],
                    'rankings': [
                        {'robot_id': r['robot_id'], 'score': r['elimination_score']}
                        for r in rankings
                    ],
                }
            )
            self.db.add(log)

        await self.db.commit()

        return {
            'action': 'eliminated',
            'eliminated_robot': eliminated,
            'all_rankings': rankings,
        }

    # ──────────────────────────────────────
    # 工具方法
    # ──────────────────────────────────────

    @staticmethod
    def _normalize(value: float, low: float, high: float) -> float:
        """线性归一化到 0~100"""
        if high == low:
            return 50
        score = (value - low) / (high - low) * 100
        return max(0, min(100, score))

    @staticmethod
    def _calc_max_drawdown(nav_list: list[float]) -> float:
        """计算最大回撤（负数）"""
        if not nav_list:
            return 0
        peak = nav_list[0]
        max_dd = 0
        for nav in nav_list:
            if nav > peak:
                peak = nav
            dd = (nav - peak) / peak
            if dd < max_dd:
                max_dd = dd
        return max_dd
```

```python
# backend/app/api/ws/realtime.py

import json
import asyncio
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends, Query
from app.core.redis_client import redis_client
import logging

logger = logging.getLogger(__name__)

router = APIRouter()


class ConnectionManager:
    """WebSocket 连接管理"""

    def __init__(self):
        # {channel: set[WebSocket]}
        self.connections: dict[str, set[WebSocket]] = {}

    async def connect(self, ws: WebSocket, channel: str):
        await ws.accept()
        if channel not in self.connections:
            self.connections[channel] = set()
        self.connections[channel].add(ws)
        logger.info(f"[WS] 连接 {channel}, 当前: {len(self.connections[channel])}")

    async def disconnect(self, ws: WebSocket, channel: str):
        self.connections.get(channel, set()).discard(ws)

    async def broadcast(self, channel: str, data: dict):
        conns = self.connections.get(channel, set())
        dead = set()
        for ws in conns:
            try:
                await ws.send_json(data)
            except Exception:
                dead.add(ws)
        for ws in dead:
            conns.discard(ws)


manager = ConnectionManager()


@router.websocket("/ws/quotes")
async def ws_quotes(ws: WebSocket, codes: str = Query("")):
    """
    实时行情推送
    客户端连接时传 ?codes=000001,600519
    服务端每秒推送最新行情
    """
    channel = f"quotes:{codes}"
    await manager.connect(ws, channel)

    stock_codes = [c.strip() for c in codes.split(",") if c.strip()]

    try:
        while True:
            # 从 Redis 读取缓存行情
            data = {}
            for code in stock_codes:
                cached = await redis_client.hgetall(f"quote:{code}")
                if cached:
                    data[code] = {
                        'price': float(cached.get('price', 0)),
                        'open': float(cached.get('open', 0)),
                        'high': float(cached.get('high', 0)),
                        'low': float(cached.get('low', 0)),
                        'prev_close': float(cached.get('prev_close', 0)),
                        'volume': int(cached.get('volume', 0)),
                    }
            if data:
                await ws.send_json({'type': 'quotes', 'data': data})
            await asyncio.sleep(1)
    except WebSocketDisconnect:
        await manager.disconnect(ws, channel)


@router.websocket("/ws/pnl/{robot_id}")
async def ws_pnl(ws: WebSocket, robot_id: int):
    """
    实时 PnL 推送
    从 Redis hash pnl:{robot_id} 读取
    """
    channel = f"pnl:{robot_id}"
    await manager.connect(ws, channel)

    try:
        while True:
            pnl_data = await redis_client.hgetall(f"pnl:{robot_id}")
            if pnl_data:
                parsed = {
                    code: float(val)
                    for code, val in pnl_data.items()
                }
                await ws.send_json({'type': 'pnl', 'robot_id': robot_id, 'data': parsed})
            await asyncio.sleep(2)
    except WebSocketDisconnect:
        await manager.disconnect(ws, channel)

    """

    await pubsub.subscribe(f"ws:robot:{robot_id}")

        while True:
            message = await pubsub.get_message(
                ignore_subscribe_messages=True, timeout=1.0
            )
            if message and message['type'] == 'message':
                data = json.loads(message['data'])
                await ws.send_json(data)
            await asyncio.sleep(0.1)
    except WebSocketDisconnect:
        await manager.disconnect(ws, channel)
        await pubsub.unsubscribe(f"ws:robot:{robot_id}")
    finally:
        await pubsub.close()
```

```python
@celery_app.task(bind=True, name="data.sync")
def sync_data(self, categories: list[str], trade_date: str = None) -> dict:
    """
    数据同步任务
    categories: ["daily_kline", "lhb_detail", ...]
    """
    from app.data_providers.manager import data_manager
    from app.data_providers.base import DataCategory

    results = {}
    total = len(categories)

    for i, cat_name in enumerate(categories):
        try:
            category = DataCategory(cat_name)
            data, provider, failed = data_manager.fetch_sync(
                category, trade_date=trade_date
            )
            results[cat_name] = {
                'status': 'ok' if data is not None else 'failed',
                'provider': provider,
                'failed_providers': failed,
                'rows': len(data) if hasattr(data, '__len__') else 0,
            }
        except Exception as e:
            results[cat_name] = {
                'status': 'error',
                'error': str(e),
            }

        self.update_state(
            state='PROGRESS',
            meta={'progress': (i + 1) / total * 100}
        )

    return results


@celery_app.task(bind=True, name="data.backfill")

    self,
    category: str,
    start_date: str,
    end_date: str
) -> dict:
    """历史数据回填"""
    from app.services.market_data import MarketDataService
    from app.utils.trade_calendar import get_trade_dates

    dates = get_trade_dates(start_date, end_date)
    total = len(dates)
    success = 0
    errors = []

    for i, td in enumerate(dates):
        try:
            # 按日期逐天回填
            MarketDataService.backfill_date(category, td)
            success += 1
        except Exception as e:
            errors.append({'date': td, 'error': str(e)})

        self.update_state(
            state='PROGRESS',
            meta={
                'progress': (i + 1) / total * 100,
                'success': success,
                'errors': len(errors),
            }
        )

    return {
        'total_dates': total,
        'success': success,
        'errors': errors,
```

```python
        robots = result.scalars().all()

                await nav_svc.write_weekly_summary(robot.id)
            except Exception as e:
                logger.error(f"[Weekly] {robot.name} 异常: {e}", exc_info=True)

        await db.commit()
```

```python
# backend/app/core/redis_client.py

import redis.asyncio as aioredis
from typing import Optional
import logging

logger = logging.getLogger(__name__)


class RedisClient:
    """Redis 异步客户端封装"""

    def __init__(self):
        self._redis: Optional[aioredis.Redis] = None

    async def initialize(self, url: str):
        self._redis = aioredis.from_url(
            url,
            encoding="utf-8",
            decode_responses=True,
            max_connections=20,
        )
        await self._redis.ping()
        logger.info(f"[Redis] 连接成功: {url}")

    async def close(self):
        if self._redis:
            await self._redis.close()
            logger.info("[Redis] 连接已关闭")

    # ── 代理方法 ──
    async def get(self, key: str) -> Optional[str]:
        return await self._redis.get(key)

    async def set(self, key: str, value: str, ex: int = None):
        await self._redis.set(key, value, ex=ex)

    async def hset(self, name: str, key: str = None, value=None, mapping=None):
        if mapping:
            await self._redis.hset(name, mapping=mapping)
        elif key and value:
            await self._redis.hset(name, key, value)

    async def hgetall(self, name: str) -> dict:
        return await self._redis.hgetall(name)

    async def expire(self, name: str, seconds: int):
        await self._redis.expire(name, seconds)

    async def publish(self, channel: str, message: str):
        await self._redis.publish(channel, message)

    def pubsub(self):
        return self._redis.pubsub()

    async def delete(self, *keys):
        await self._redis.delete(*keys)


redis_client = RedisClient()
```

```python
        rotation="00:00",      # 每天零点轮转

        compression="gz",      # 压缩旧日志
        encoding="utf-8",
        format="{time:YYYY-MM-DD HH:mm:ss.SSS} | {level: <8} | "

    # 错误日志单独文件

        f"{settings.LOG_DIR}/error_{{time:YYYY-MM-DD}}.log",
        level="ERROR",

        retention="60 days",
```

```python
        monday = date(2026, 4, 13)     # 下周一

        trade_days = count_trade_days(friday, monday)
        natural_days = (monday - friday).days

        assert trade_days == 1
        assert natural_days == 3
```

## 十五、关键约束清单

```
┌──────────────────────────────────────────────────────────────────┐
│                    关键约束（开发过程中必须遵守）                    │
├──────────────────────────────────────────────────────────────────┤
│                                                                  │
│  1. 数据一致性                                                    │
│     ├── robot_trades + cash_ledger + robots.current_cash         │
│     │   三者必须在同一事务内更新                                    │
│     ├── async with db.begin(): 包裹所有交易操作                    │
│     └── cash_ledger 可由 amount 累加反推验证                      │
│                                                                  │
│  2. 防未来函数                                                    │
│     ├── 回测 K 线严格截断到 trade_date                             │
│     ├── T+N 价格必须走交易日历，禁止自然日偏移                      │
│     └── 实盘评分截断到当日                                         │
│                                                                  │
│  3. 费用不可省略                                                  │
│     ├── 回测必须包含双边交易成本                                    │
│     ├── 实盘每笔买卖必须写 cash_ledger                            │
│     └── 佣金最低 5 元规则不可忽略                                  │
│                                                                  │
│  4. audit_log 永不删除                                            │
│     ├── 数据库触发器阻止 DELETE                                    │
│     └── 其余表可以 TRUNCATE 重建                                  │
│                                                                  │
│  5. 评分引擎唯一实现                                               │
│     ├── 全系统所有评分必须经 services/scorer.py                    │
│     └── 禁止在 pages/ 或其他模块中自行计算评分                     │
│                                                                  │
│  6. 数据源降级链                                                  │
│     ├── 任意节点失败 → 自动降级下一节点                             │
│     ├── 全部失败 → 返回 None + 标记失败 providers                 │
│     └── 不可因单数据源故障导致系统崩溃                              │
│                                                                  │
│  7. 写入模式                                                      │
│     ├── 所有数据表使用 upsert (ON CONFLICT DO UPDATE)             │
│     ├── 禁止裸 INSERT 导致重复                                    │
│     └── UNIQUE 约束是最后防线                                     │
│                                                                  │
│  8. 前端纯展示                                                    │
│     ├── Vue 组件禁止直接查数据库                                   │
│     ├── 所有数据通过 REST API / WebSocket 获取                    │
│     └── 业务逻辑全部在后端 services/ 层                            │
│                                                                  │
│  9. 净值计算                                                      │
│     ├── 持仓市值用前复权当日收盘价                                  │
│     ├── 前端标注「净值基于前复权价计算」                             │
│     └── 除权日必须处理                                             │
│                                                                  │
│ 10. 部署                                                          │
│     ├── docker-compose up -d 一键部署                             │
│     ├── 环境变量通过 .env 管理                                    │
│     └── 数据库迁移用 alembic upgrade head                         │
│                                                                  │
└──────────────────────────────────────────────────────────────────┘
```

- **前端 Vue3 + TS**，15个页面，3类图表库
- **6个阶段**，总工期约 10~12 周
