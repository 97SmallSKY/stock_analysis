# 多数据源重构方案设计

## 一、现状问题分析

```
现有问题:
┌─────────────────────────────────────────────────────┐
│ 1. data_fetcher.py 职责混乱 - 数据获取+业务逻辑耦合  │
│ 2. 降级逻辑散落各处 - 每个函数各自实现retry         │
│ 3. 数据源优先级硬编码 - 无法运行时动态调整           │

```

```
         请求失败率 > 阈值
```

### 5.2 优先级配置系统

```python
# data_sources/priority_config.py

from __future__ import annotations
from dataclasses import dataclass, field, asdict
from typing import Dict, List, Optional
from enum import Enum
import json
import logging
from datetime import datetime

logger = logging.getLogger(__name__)


class DataSource(str, Enum):
    TUSHARE  = "tushare"
    AKSHARE  = "akshare"
    SINA     = "sina"
    CRAWLER  = "crawler"


class DataCategory(str, Enum):
    DAILY_KLINE      = "daily_kline"       # 日K数据
    LHB_DETAIL       = "lhb_detail"        # 龙虎榜明细
    LHB_BROKER       = "lhb_broker"        # 龙虎榜营业部
    INSTITUTION      = "institution"        # 机构数据
    REALTIME_QUOTE   = "realtime_quote"    # 实时行情
    MARKET_SENTIMENT = "market_sentiment"  # 市场情绪
    FINANCIAL_DATA   = "financial_data"    # 财务数据
    BLOCK_TRADE      = "block_trade"       # 大宗交易


# ── 默认优先级表(业务语义写死, 但可被DB覆盖) ─────────────────────────
DEFAULT_PRIORITY: Dict[DataCategory, List[DataSource]] = {
    DataCategory.DAILY_KLINE: [
        DataSource.TUSHARE,
        DataSource.AKSHARE,
        DataSource.SINA,
        DataSource.CRAWLER,
    ],
    DataCategory.LHB_DETAIL: [
        DataSource.AKSHARE,
        DataSource.SINA,
        DataSource.CRAWLER,
    ],
    DataCategory.LHB_BROKER: [
        DataSource.AKSHARE,
        DataSource.CRAWLER,
    ],
    DataCategory.INSTITUTION: [
        DataSource.SINA,
        DataSource.CRAWLER,
    ],
    DataCategory.REALTIME_QUOTE: [
        DataSource.AKSHARE,
        DataSource.SINA,
    ],
    DataCategory.MARKET_SENTIMENT: [
        DataSource.AKSHARE,
        DataSource.SINA,
    ],
    DataCategory.FINANCIAL_DATA: [
        DataSource.TUSHARE,
        DataSource.AKSHARE,
    ],
    DataCategory.BLOCK_TRADE: [
        DataSource.AKSHARE,
        DataSource.CRAWLER,
    ],
}


@dataclass
class SourceMeta:
    """单个数据源的运行时元数据"""
    source: DataSource
    enabled: bool = True
    rate_limit: int = 60          # 次/分钟
    timeout: int = 30             # 秒
    max_retries: int = 3
    base_delay: float = 2.0       # 重试初始等待
    credentials: Dict = field(default_factory=dict)
    note: str = ""
    updated_at: str = ""

    def to_dict(self) -> dict:
        d = asdict(self)
        d["source"] = self.source.value
        return d

    @classmethod
    def from_dict(cls, d: dict) -> "SourceMeta":
        d = d.copy()
        d["source"] = DataSource(d["source"])
        return cls(**d)


class PriorityConfig:
    """
    优先级配置管理器
    - 内存中维护当前优先级表
    - 支持从 DB / JSON 文件持久化加载与保存
    - 提供增删改查接口供 UI 调用
    """

    _TABLE = "data_source_config"      # PostgreSQL 表名
    _JSON_PATH = "data/ds_config.json" # Excel模式备用

    def __init__(self, storage):
        self._storage = storage
        # category -> 有序 DataSource 列表
        self._priority: Dict[DataCategory, List[DataSource]] = {}
        # source -> 元数据
        self._meta: Dict[DataSource, SourceMeta] = {}
        self._load()

    # ── 加载 ─────────────────────────────────────────────────────────
    def _load(self):
        """从持久化存储加载, 不存在则使用默认值"""
        try:
            raw = self._load_from_storage()
            if raw:
                self._priority = {
                    DataCategory(k): [DataSource(s) for s in v["order"]]
                    for k, v in raw.get("priority", {}).items()
                }
                self._meta = {
                    DataSource(k): SourceMeta.from_dict(v)
                    for k, v in raw.get("meta", {}).items()
                }
                logger.info("✅ 数据源配置从存储加载成功")
                return
        except Exception as e:
            logger.warning(f"加载数据源配置失败, 使用默认值: {e}")

        # 使用默认值初始化
        self._priority = {k: list(v) for k, v in DEFAULT_PRIORITY.items()}
        self._meta = {
            DataSource.TUSHARE: SourceMeta(DataSource.TUSHARE, rate_limit=200, max_retries=3),
            DataSource.AKSHARE: SourceMeta(DataSource.AKSHARE, rate_limit=100, max_retries=5),
            DataSource.SINA:    SourceMeta(DataSource.SINA,    rate_limit=60,  max_retries=3),
            DataSource.CRAWLER: SourceMeta(DataSource.CRAWLER, rate_limit=20,  max_retries=2),
        }

    def _load_from_storage(self) -> Optional[dict]:
        import os, json
        if os.path.exists(self._JSON_PATH):
            with open(self._JSON_PATH) as f:
                return json.load(f)
        return None

    # ── 保存 ─────────────────────────────────────────────────────────
    def save(self):
        """持久化当前配置"""
        payload = {
            "priority": {
                k.value: {"order": [s.value for s in v]}
                for k, v in self._priority.items()
            },
            "meta": {
                k.value: v.to_dict()
                for k, v in self._meta.items()
            },
            "saved_at": datetime.now().isoformat(),
        }
        import json, os
        os.makedirs("data", exist_ok=True)
        with open(self._JSON_PATH, "w") as f:
            json.dump(payload, f, ensure_ascii=False, indent=2)
        logger.info("✅ 数据源配置已持久化")

    # ── 查询接口 ─────────────────────────────────────────────────────
    def get_ordered_sources(
        self, category: DataCategory
    ) -> List[DataSource]:
        """返回指定类别的启用数据源列表(按优先级)"""
        all_sources = self._priority.get(category, DEFAULT_PRIORITY.get(category, []))
        return [s for s in all_sources if self._meta.get(s, SourceMeta(s)).enabled]

    def get_meta(self, source: DataSource) -> SourceMeta:
        return self._meta.get(source, SourceMeta(source))
    # ── 修改接口(供UI调用) ────────────────────────────────────────────
    def set_order(
        self,
        category: DataCategory,
        ordered_sources: List[DataSource],
    ):
        """重新设置某类别的优先级顺序"""
        self._priority[category] = ordered_sources
        self.save()
        logger.info(f"优先级更新 [{category.value}]: {[s.value for s in ordered_sources]}")

    def move_up(self, category: DataCategory, source: DataSource):
        """将某数据源在指定类别中上移一位"""
        order = self._priority.get(category, [])
        idx = order.index(source) if source in order else -1
        if idx > 0:
            order[idx], order[idx - 1] = order[idx - 1], order[idx]
            self._priority[category] = order
            self.save()

    def move_down(self, category: DataCategory, source: DataSource):
        """将某数据源在指定类别中下移一位"""
        order = self._priority.get(category, [])
        idx = order.index(source) if source in order else -1
        if idx != -1 and idx < len(order) - 1:
            order[idx], order[idx + 1] = order[idx + 1], order[idx]
            self._priority[category] = order
            self.save()

    def toggle_source(self, source: DataSource, enabled: bool):
        """启用/禁用某数据源(全局生效)"""
        if source in self._meta:
            self._meta[source].enabled = enabled
            self._meta[source].updated_at = datetime.now().isoformat()
            self.save()

    def update_meta(self, source: DataSource, **kwargs):
        """更新数据源元数据: rate_limit/timeout/credentials等"""
        if source in self._meta:
            for k, v in kwargs.items():
                if hasattr(self._meta[source], k):
                    setattr(self._meta[source], k, v)
            self._meta[source].updated_at = datetime.now().isoformat()
            self.save()

    def reset_to_default(self, category: Optional[DataCategory] = None):
        """重置为默认优先级"""
        if category:
            self._priority[category] = list(DEFAULT_PRIORITY.get(category, []))
        else:
            self._priority = {k: list(v) for k, v in DEFAULT_PRIORITY.items()}
        self.save()

    def get_all_config(self) -> dict:
        """返回全量配置快照(供UI展示)"""
        return {
            "priority": {
                cat.value: [s.value for s in sources]
                for cat, sources in self._priority.items()
            },
            "meta": {
                src.value: self._meta[src].to_dict()
                for src in self._meta
            },
        }
```

### 5.3 熔断器实现

```python
# data_sources/circuit_breaker.py

from __future__ import annotations
import time
import threading
from dataclasses import dataclass, field
from typing import Dict, Optional
from enum import Enum
import logging

logger = logging.getLogger(__name__)


class CBState(str, Enum):
    CLOSED    = "closed"      # 正常通行
    OPEN      = "open"        # 熔断中, 拒绝请求
    HALF_OPEN = "half_open"   # 探测中, 放行一个请求


@dataclass
class CBConfig:
    failure_threshold: int   = 5      # 连续失败N次后熔断
    success_threshold: int   = 2      # 半开状态成功N次后恢复
    reset_timeout: float     = 60.0   # 熔断后多少秒进入半开
    failure_rate_threshold: float = 0.5  # 失败率阈值
    window_size: int         = 10     # 统计窗口(最近N次请求)


@dataclass
class CBStatus:
    state: CBState               = CBState.CLOSED
    failure_count: int           = 0
    success_count: int           = 0
    last_failure_time: float     = 0.0
    last_state_change: float     = field(default_factory=time.time)
    # 滑动窗口: True=成功 False=失败
    window: list                 = field(default_factory=list)
    total_requests: int          = 0
    total_failures: int          = 0

    def failure_rate(self) -> float:
        if not self.window:
            return 0.0
        return self.window.count(False) / len(self.window)

    def record(self, success: bool, window_size: int):
        self.window.append(success)
        if len(self.window) > window_size:
            self.window.pop(0)
        self.total_requests += 1
        if not success:
            self.total_failures += 1


class CircuitBreaker:
    """
    每个 (DataSource, DataCategory) 独立维护一个熔断器
    线程安全
    """

    def __init__(self, config: CBConfig = None):
        self._config = config or CBConfig()
        # key: (source, category) -> CBStatus
        self._status: Dict[tuple, CBStatus] = {}
        self._lock = threading.Lock()

    def _key(self, source: str, category: str) -> tuple:
        return (source, category)

    def _get_status(self, source: str, category: str) -> CBStatus:
        k = self._key(source, category)
        if k not in self._status:
            self._status[k] = CBStatus()
        return self._status[k]

    def allow_request(self, source: str, category: str) -> bool:
        """判断是否允许本次请求通过"""
        with self._lock:
            st = self._get_status(source, category)

            if st.state == CBState.CLOSED:
                return True

            if st.state == CBState.OPEN:
                # 判断是否到了半开探测时间
                if time.time() - st.last_failure_time >= self._config.reset_timeout:
                    st.state = CBState.HALF_OPEN
                    st.success_count = 0
                    logger.info(f"🔶 熔断器半开探测 [{source}:{category}]")
                    return True
                return False   # 熔断中, 直接拒绝

            if st.state == CBState.HALF_OPEN:
                return True    # 半开状态放行探测请求

        return True

    def on_success(self, source: str, category: str):
        """请求成功回调"""
        with self._lock:
            st = self._get_status(source, category)
            st.record(True, self._config.window_size)

            if st.state == CBState.HALF_OPEN:
                st.success_count += 1
                if st.success_count >= self._config.success_threshold:
                    st.state = CBState.CLOSED
                    st.failure_count = 0
                    logger.info(f"✅ 熔断器恢复 [{source}:{category}]")

            elif st.state == CBState.CLOSED:
                st.failure_count = 0  # 成功则重置连续失败计数

    def on_failure(self, source: str, category: str):
        """请求失败回调"""
        with self._lock:
            st = self._get_status(source, category)
            st.record(False, self._config.window_size)
            st.failure_count += 1
            st.last_failure_time = time.time()

            should_open = (
                st.failure_count >= self._config.failure_threshold
                or st.failure_rate() >= self._config.failure_rate_threshold
            )

            if should_open and st.state != CBState.OPEN:
                st.state = CBState.OPEN
                st.last_state_change = time.time()
                logger.warning(
                    f"🔴 熔断器开启 [{source}:{category}] "
                    f"连续失败={st.failure_count} "
                    f"失败率={st.failure_rate():.1%}"
                )

            elif st.state == CBState.HALF_OPEN:
                # 半开探测失败, 回到熔断
                st.state = CBState.OPEN
                logger.warning(f"🔴 熔断器探测失败, 重新熔断 [{source}:{category}]")

    def get_all_status(self) -> Dict[tuple, dict]:
        """返回全量状态快照(供UI展示)"""
        with self._lock:
            result = {}
            for (src, cat), st in self._status.items():
                result[(src, cat)] = {
                    "state":          st.state.value,
                    "failure_count":  st.failure_count,
                    "failure_rate":   f"{st.failure_rate():.1%}",
                    "total_requests": st.total_requests,
                    "total_failures": st.total_failures,
                }
            return result

    def force_reset(self, source: str, category: str):
        """手动重置熔断器(供UI紧急操作)"""
        with self._lock:
            k = self._key(source, category)
            self._status[k] = CBStatus()
            logger.info(f"🔧 熔断器手动重置 [{source}:{category}]")
```

### 5.4 统一缓存管理器

```python
# data_sources/cache_manager.py

from __future__ import annotations
import time
import hashlib
import json
import threading
from typing import Any, Optional, Dict, Tuple
from dataclasses import dataclass
import pandas as pd
import logging

logger = logging.getLogger(__name__)


@dataclass
class CacheEntry:
    data: Any
    created_at: float
    ttl: float          # 秒, -1 表示永不过期
    source: str         # 是哪个数据源产生的
    hit_count: int = 0

    def is_expired(self) -> bool:
        if self.ttl < 0:
            return False
        return time.time() - self.created_at > self.ttl

    def age_seconds(self) -> float:
        return time.time() - self.created_at


# ── TTL 策略(按数据类别) ─────────────────────────────────────────────
DEFAULT_TTL: Dict[str, float] = {
    "daily_kline":      3600 * 6,    # 日K: 6小时
    "lhb_detail":       3600 * 24,   # 龙虎榜: 24小时
    "lhb_broker":       3600 * 24,
    "institution":      3600 * 24,
    "realtime_quote":   12,          # 实时行情: 12秒
    "market_sentiment": 300,         # 市场情绪: 5分钟
    "financial_data":   3600 * 24 * 7,
    "block_trade":      3600 * 12,
    "default":          300,
}


class CacheManager:
    """
    统一内存缓存管理器(线程安全)
    后续可替换为 Redis 后端
    """

    def __init__(self):
        self._store: Dict[str, CacheEntry] = {}
        self._lock = threading.Lock()
        self._stats = {"hits": 0, "misses": 0, "evictions": 0}

    # ── Key构造 ──────────────────────────────────────────────────────
    @staticmethod
    def build_key(category: str, **params) -> str:
        """根据类别+参数生成稳定缓存Key"""
        param_str = json.dumps(params, sort_keys=True, default=str)
        h = hashlib.md5(param_str.encode()).hexdigest()[:8]
        return f"{category}:{h}"

    # ── 读写 ─────────────────────────────────────────────────────────
    def get(self, key: str) -> Tuple[bool, Optional[Any], Optional[str]]:
        """
        返回 (命中, 数据, 来源数据源)
        """
        with self._lock:
            entry = self._store.get(key)
            if entry is None or entry.is_expired():
                if entry:
                    del self._store[key]
                    self._stats["evictions"] += 1
                self._stats["misses"] += 1
                return False, None, None
            entry.hit_count += 1
            self._stats["hits"] += 1
            return True, entry.data, entry.source

    def set(
        self,
        key: str,
        data: Any,
        category: str,
        source: str,
        ttl: Optional[float] = None,
    ):
        ttl = ttl if ttl is not None else DEFAULT_TTL.get(category, DEFAULT_TTL["default"])
        with self._lock:
            self._store[key] = CacheEntry(
                data=data,
                created_at=time.time(),
                ttl=ttl,
                source=source,
            )

    def invalidate(self, pattern: str = ""):
        """清除匹配前缀的缓存"""
        with self._lock:
            keys = [k for k in self._store if k.startswith(pattern)]
            for k in keys:
                del self._store[k]
            logger.info(f"缓存清除: {len(keys)} 条 (pattern={pattern!r})")

    def get_stats(self) -> dict:
        with self._lock:
            total = self._stats["hits"] + self._stats["misses"]
            hit_rate = self._stats["hits"] / total if total > 0 else 0
            return {
                **self._stats,
                "total":    total,
                "hit_rate": f"{hit_rate:.1%}",
                "size":     len(self._store),
            }

    def get_entries_info(self) -> list:
        """返回所有缓存条目信息(供UI展示)"""
        with self._lock:
            result = []
            for key, entry in self._store.items():
                result.append({
                    "key":       key,
                    "source":    entry.source,
                    "age":       f"{entry.age_seconds():.0f}s",
                    "ttl":       f"{entry.ttl:.0f}s" if entry.ttl > 0 else "永久",
                    "expired":   entry.is_expired(),
                    "hit_count": entry.hit_count,
                    "size":      (
                        f"{len(entry.data)}行"
                        if isinstance(entry.data, pd.DataFrame)
                        else "N/A"
                    ),
                })
            return result
```

```python
# data_sources/adapters/base.py

from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Optional, Any
import pandas as pd
import logging


@dataclass
class FetchResult:
    """统一返回结构 - 所有适配器必须返回此对象"""
    success:     bool
    data:        Optional[pd.DataFrame]
    source:      str                        # 实际来源数据源名称
    category:    str                        # 数据类别
    from_cache:  bool      = False          # 是否来自缓存
    from_fallback: bool    = False          # 是否为降级数据
    error:       str       = ""             # 失败原因
    extra:       dict      = field(default_factory=dict)   # 附加信息

    @classmethod
    def ok(cls, data: pd.DataFrame, source: str, category: str, **kwargs) -> "FetchResult":
        return cls(success=True, data=data, source=source, category=category, **kwargs)

    @classmethod
    def fail(cls, source: str, category: str, error: str = "") -> "FetchResult":
        return cls(success=False, data=None, source=source, category=category, error=error)


class BaseDataAdapter(ABC):
    """
    所有数据源适配器抽象基类
    - 子类只需实现具体的 fetch_xxx 方法
    - 统一出参格式由 FetchResult 保证
    - 字段标准化由各子类 _normalize_xxx 完成
    """

    def __init__(self, meta):
        self.meta   = meta
        self.logger = logging.getLogger(self.__class__.__name__)

    # ── 必须实现的接口 ────────────────────────────────────────────────

    @abstractmethod
    async def fetch_daily_kline(
        self,
        stock_code: str,
        start_date: str,   # YYYY-MM-DD
        end_date:   str,
    ) -> FetchResult:
        """
        日K数据 - 统一输出列:
        trade_date | open | high | low | close |
        volume | amount | pct_chg | turnover_rate
        """

    @abstractmethod
    async def fetch_lhb_detail(
        self,
        trade_date: str,
    ) -> FetchResult:
        """
        龙虎榜每日明细 - 统一输出列:
        trade_date | stock_code | stock_name | close_price |
        change_pct | lhb_reason | total_buy  | total_sell  |
        net_buy    | buy_pct    | inst_buy   | inst_sell
        """

    @abstractmethod
    async def fetch_lhb_broker(
        self,
        stock_code: str,
        trade_date: str,
    ) -> FetchResult:
        """
        龙虎榜营业部明细 - 统一输出列:
        trade_date | stock_code | direction | rank |
        broker_name | amount | is_institution | pct_of_total
        """

    @abstractmethod
    async def fetch_institution(
        self,
        stock_code: str,
        start_date: str,
        end_date:   str,
    ) -> FetchResult:
        """
        机构持仓/席位数据 - 统一输出列:
        trade_date | stock_code | institution_name |
        hold_ratio | hold_change | institution_type
        """

    @abstractmethod
    async def health_check(self) -> bool:
        """数据源健康检查, 返回 True 表示可用"""

    # ── 通用工具方法 ──────────────────────────────────────────────────

    @staticmethod
    def _safe_float(val: Any, default: float = 0.0) -> float:
        try:
            return float(val)
        except (TypeError, ValueError):
            return default

    @staticmethod
    def _normalize_stock_code(code: str) -> str:
        """统一股票代码格式: 000001.SZ / 600000.SH"""
        code = str(code).strip()
        if "." in code:
            return code.upper()
        if code.startswith(("6", "5")):
            return f"{code}.SH"
        return f"{code}.SZ"

    def _ensure_columns(
        self,
        df: pd.DataFrame,
        required: list,
        fill_value: Any = 0,
    ) -> pd.DataFrame:
        """确保输出DataFrame包含所有必要列"""
        for col in required:
            if col not in df.columns:
                self.logger.debug(f"补充缺失列: {col}")
                df[col] = fill_value
        return df
```

```python
        except Exception as e:

    async def _run(self, func, *args, **kwargs):
        """同步转异步执行"""
        loop = asyncio.get_event_loop()

    # ── 日K数据(核心) ─────────────────────────────────────────────────
    async def fetch_daily_kline(
        self,
        stock_code: str,
        start_date: str,
        end_date:   str,
    ) -> FetchResult:
        if not self._pro:
            return FetchResult.fail("tushare", "daily_kline", "客户端未初始化")

        try:
            # TuShare 需要带后缀的代码: 000001.SZ
            ts_code = self._normalize_stock_code(stock_code)
            start   = start_date.replace("-", "")
            end     = end_date.replace("-", "")

            raw = await self._run(
                self._pro.daily,
                ts_code=ts_code,
                start_date=start,
                end_date=end,
            )

            if raw is None or raw.empty:
                return FetchResult.fail("tushare", "daily_kline", "返回空数据")

            # 同时获取换手率
            daily_basic = await self._run(
                self._pro.daily_basic,
                ts_code=ts_code,
                start_date=start,
                end_date=end,
                fields="trade_date,turnover_rate,volume_ratio",
            )

            df = self._normalize_kline(raw, daily_basic)
            return FetchResult.ok(df, "tushare", "daily_kline")

        except Exception as e:
            logger.error(f"TuShare fetch_daily_kline 失败: {e}")
            return FetchResult.fail("tushare", "daily_kline", str(e))

    def _normalize_kline(
        self,
        raw: pd.DataFrame,
        basic: Optional[pd.DataFrame],
    ) -> pd.DataFrame:
        df = raw.rename(columns=_KLINE_COL_MAP).copy()

        # trade_date: '20240115' -> '2024-01-15'
        df["trade_date"] = pd.to_datetime(
            df["trade_date"], format="%Y%m%d"
        ).dt.strftime("%Y-%m-%d")

        # 合并换手率
        if basic is not None and not basic.empty:
            basic["trade_date"] = pd.to_datetime(
                basic["trade_date"], format="%Y%m%d"
            ).dt.strftime("%Y-%m-%d")
            df = df.merge(
                basic[["trade_date", "turnover_rate", "volume_ratio"]],
                on="trade_date",
                how="left",
                suffixes=("", "_basic"),
            )
            # 优先使用 daily_basic 的换手率
            if "turnover_rate_basic" in df.columns:
                df["turnover_rate"] = df["turnover_rate_basic"].fillna(
                    df.get("turnover_rate", 0)
                )
                df.drop(columns=["turnover_rate_basic"], inplace=True)

        # 统一数值类型
        numeric_cols = ["open", "high", "low", "close", "volume",
                        "amount", "pct_chg", "turnover_rate"]
        for col in numeric_cols:
            if col in df.columns:
                df[col] = pd.to_numeric(df[col], errors="coerce").fillna(0)

        return df.sort_values("trade_date").reset_index(drop=True)

    # ── 龙虎榜(TuShare有接口但非优先) ────────────────────────────────
    async def fetch_lhb_detail(self, trade_date: str) -> FetchResult:

        return self._ensure_columns(
```

```python
from __future__ import annotations
import asyncio
import re
import time
import urllib.request
from functools import partial
from typing import Optional
import pandas as pd

from .base import BaseDataAdapter, FetchResult


class SinaAdapter(BaseDataAdapter):
    """
    新浪财经适配器
    职责: 机构数据(最高优先级) + 实时行情备用 + 日K备用
    """

    _INST_URL  = "https://vip.stock.finance.sina.com.cn/q/go.php/vLHBData/kind/jgmx/index.phtml"
    _QUOTE_URL = "https://hq.sinajs.cn/list={}"

    async def _run(self, func, *args, **kwargs):
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(None, partial(func, *args, **kwargs))

    # ── 实时行情(代理感知) ─────────────────────────────────────────────
    def _fetch_sina_quote_sync(self, stock_code: str) -> Optional[dict]:
        """
        两路保障:
        1. 走系统代理
        2. 降级绕过代理直连
        """
        code = stock_code.split(".")[0]
        prefix = "sh" if code.startswith("6") else "sz"
        symbol = f"{prefix}{code}"
        url    = self._QUOTE_URL.format(symbol)

        for use_proxy in [True, False]:
            try:
                if use_proxy:
                    opener = urllib.request.build_opener()
                else:
                    opener = urllib.request.build_opener(
                        urllib.request.ProxyHandler({})
                    )
                opener.addheaders = [
                    ("Referer", "https://finance.sina.com.cn"),
                    ("User-Agent", "Mozilla/5.0"),
                ]
                with opener.open(url, timeout=5) as resp:
                    raw = resp.read().decode("gbk", errors="replace")

                match = re.search(r'"([^"]+)"', raw)
                if not match:
                    continue

                fields = match.group(1).split(",")
                if len(fields) < 32 or float(fields[3]) <= 0:
                    continue

                return {
                    "name":        fields[0],
                    "open":        float(fields[1]),
                    "prev_close":  float(fields[2]),
                    "price":       float(fields[3]),
                    "high":        float(fields[4]),
                    "low":         float(fields[5]),
                    "volume":      float(fields[8]),
                    "amount":      float(fields[9]),
                    "date":        fields[30],
                    "time":        fields[31],
                    "via_proxy":   use_proxy,
                }
            except Exception:
                continue
        return None
```

```python
    def _random_headers(self) -> Dict:
        h = random.choice(self._HEADERS_POOL).copy()
        h["Referer"] = "https://data.eastmoney.com/stock/lhb.html"
        return h

    async def _get(
        self,
        url: str,
        retries: int = 3,
        delay: float = 2.0,
    ) -> Optional[dict]:
        """带指数退避的异步GET"""
        for attempt in range(retries):
            try:
                async with aiohttp.ClientSession() as session:
                    async with session.get(
                        url,
                        headers=self._random_headers(),
                        timeout=aiohttp.ClientTimeout(total=self.meta.timeout),
                        ssl=False,
                    ) as resp:
                        if resp.status == 200:
                            text = await resp.text(encoding="utf-8", errors="replace")
                            return json.loads(text)
                        elif resp.status == 429:
                            wait = delay * (2 ** attempt) + random.uniform(0, 1)
                            self.logger.warning(f"限流429, 等待{wait:.1f}s")
                            await asyncio.sleep(wait)
            except asyncio.TimeoutError:
                wait = delay * (2 ** attempt)
                self.logger.warning(f"超时 attempt={attempt+1}, 等待{wait:.1f}s")
                await asyncio.sleep(wait)
            except Exception as e:
                self.logger.error(f"请求失败 attempt={attempt+1}: {e}")
                await asyncio.sleep(delay)
        return None

    # ── 龙虎榜明细(分页全量拉取) ──────────────────────────────────────
    async def fetch_lhb_detail(self, trade_date: str) -> FetchResult:
        date_str = trade_date.replace("-", "")
        all_records = []
        page = 1

        while True:
            url  = self._EM_LHB_URL.format(
                page=page, size=50, date=date_str
            )
            data = await self._get(url)

            if not data:
                break

            # 东方财富返回结构: data.data.data
            records = (
                data.get("data", {})
                    .get("data", {})
                    .get("data", [])
            )
            if not records:
                break

            all_records.extend(records)

            # 翻页判断
            total = (
                data.get("data", {})
                    .get("data", {})
                    .get("count", 0)
            )
            if len(all_records) >= total:
                break
            page += 1
            await asyncio.sleep(random.uniform(0.5, 1.2))

        if not all_records:
            return FetchResult.fail("crawler", "lhb_detail", "爬取结果为空")

        df = self._normalize_em_lhb(all_records, trade_date)
        return FetchResult.ok(
            df, "crawler", "lhb_detail",
            from_fallback=True,
            extra={"total_pages": page, "total_records": len(all_records)},
        )
```

```python
# data_sources/router.py

from __future__ import annotations
import asyncio
import logging
from typing import Any, Callable, Coroutine, Optional

from .priority_config import DataCategory, DataSource, PriorityConfig
from .circuit_breaker import CircuitBreaker
from .cache_manager   import CacheManager
from .adapters.base   import FetchResult

logger = logging.getLogger(__name__)


class DataRouter:
    """
    核心路由分发器
    - 按优先级依次尝试各数据源
    - 每个数据源受熔断器保护
    - 命中缓存直接返回
    - 全部失败返回 FetchResult.fail
    """

    def __init__(
        self,
        priority_config: PriorityConfig,
        circuit_breaker: CircuitBreaker,
        cache_manager:   CacheManager,
        adapters:        dict,          # DataSource -> BaseDataAdapter
    ):
        self._pc  = priority_config
        self._cb  = circuit_breaker
        self._cm  = cache_manager
        self._ads = adapters

    async def route(
        self,
        category:  DataCategory,
        method:    str,             # 适配器方法名, 如 "fetch_daily_kline"
        cache_key: str,
        **kwargs,
    ) -> FetchResult:
        """
        统一路由入口

        流程:
        1. 查缓存 → 命中直接返回
        2. 按优先级遍历数据源
            a. 熔断器检查 → 跳过熔断中的数据源
            b. 调用适配器
            c. 成功 → 写缓存 + 熔断器成功回调 + 返回
            d. 失败 → 熔断器失败回调 + 继续下一个
        3. 全部失败 → 返回 fail
        """

        # ── Step1: 缓存命中 ───────────────────────────────────────────
        hit, cached_data, cached_source = self._cm.get(cache_key)
        if hit:
            logger.debug(f"缓存命中 [{category.value}] key={cache_key}")
            return FetchResult.ok(
                cached_data,
                source=cached_source,
                category=category.value,
                from_cache=True,
            )

        # ── Step2: 按优先级遍历 ───────────────────────────────────────
        sources = self._pc.get_ordered_sources(category)
        if not sources:
            return FetchResult.fail("none", category.value, "无可用数据源")

        last_error = ""
        tried = []

        for source in sources:
            adapter = self._ads.get(source)
            if adapter is None:
                logger.debug(f"适配器未注册: {source.value}")
                continue

            # 熔断器检查
            if not self._cb.allow_request(source.value, category.value):
                logger.debug(f"熔断器拒绝 [{source.value}:{category.value}]")
                continue

            tried.append(source.value)

            try:
                func: Callable[..., Coroutine] = getattr(adapter, method)
                result: FetchResult = await asyncio.wait_for(
                    func(**kwargs),
                    timeout=adapter.meta.timeout,
                )
            except asyncio.TimeoutError:
                last_error = f"{source.value} 超时"
                self._cb.on_failure(source.value, category.value)
                logger.warning(f"⏱ 超时 [{source.value}:{category.value}]")
                continue
            except Exception as e:
                last_error = str(e)
                self._cb.on_failure(source.value, category.value)
                logger.error(f"❌ 异常 [{source.value}:{category.value}]: {e}")
                continue

            if result.success and result.data is not None and not result.data.empty:
                # 成功: 写缓存 + 熔断器回调
                self._cb.on_success(source.value, category.value)
                self._cm.set(
                    cache_key,
                    result.data,
                    category=category.value,
                    source=source.value,
                )
                if len(tried) > 1:
                    logger.info(
                        f"⚡ 降级成功 [{category.value}] "
                        f"tried={tried} final={source.value}"
                    )
                return result
            else:
                # 数据为空也算失败, 触发降级
                last_error = result.error or "返回空数据"
                self._cb.on_failure(source.value, category.value)
                logger.warning(
                    f"⚠ 数据源返回空 [{source.value}:{category.value}] "
                    f"error={last_error}"
                )
                continue

        # ── Step3: 全部失败 ───────────────────────────────────────────
        logger.error(
            f"🚨 所有数据源失败 [{category.value}] "
            f"tried={tried} last_error={last_error}"
        )
        return FetchResult.fail("none", category.value, f"全部失败: {last_error}")
```

```python
        dsm = DataSourceManager.get_instance()

        if result.success:

    _instance: Optional["DataSourceManager"] = None
    _lock = threading.Lock()

    # ── 单例工厂 ──────────────────────────────────────────────────────
    @classmethod
    def get_instance(cls) -> "DataSourceManager":
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = cls._create()
        return cls._instance

    @classmethod
    def _create(cls) -> "DataSourceManager":
        from utils.storage import get_storage
        storage = get_storage()
        return cls(storage)

    def reset_instance(cls):
        """测试用: 重置单例"""
        with cls._lock:
            cls._instance = None

            self.fetch_institution(stock_code, start_date, end_date)
        )

    @staticmethod
    def _run_async(coro) -> FetchResult:
        """在任意线程中运行异步方法"""
        try:
            loop = asyncio.get_event_loop()
            if loop.is_running():
                # Streamlit 环境: 已有事件循环
                import concurrent.futures
                with concurrent.futures.ThreadPoolExecutor() as pool:
                    future = pool.submit(asyncio.run, coro)
                    return future.result()
            else:
                return loop.run_until_complete(coro)
        except Exception as e:
            logger.error(f"_run_async 失败: {e}")
            return FetchResult.fail("none", "unknown", str(e))
```

### 5.8 健康检查器

```python
# data_sources/health_checker.py

from __future__ import annotations
import asyncio
import time
import logging
from typing import Dict
from .circuit_breaker import CircuitBreaker

logger = logging.getLogger(__name__)


class HealthChecker:
    """
    定期探测各数据源可用性
    结果写回熔断器状态
    """

    def __init__(self, adapters: dict, cb: CircuitBreaker):
        self._adapters = adapters
        self._cb       = cb
        self._results: Dict[str, dict] = {}

    async def check_all(self) -> Dict[str, dict]:
        """并发检查所有数据源"""
        tasks = {
            source.value: asyncio.create_task(
                self._check_one(source.value, adapter)
            )
            for source, adapter in self._adapters.items()
        }
        results = await asyncio.gather(*tasks.values(), return_exceptions=True)

        self._results = {}
        for source_name, result in zip(tasks.keys(), results):
            if isinstance(result, Exception):
                self._results[source_name] = {
                    "healthy": False,
                    "latency_ms": -1,
                    "error": str(result),
                    "checked_at": time.strftime("%H:%M:%S"),
                }
            else:
                self._results[source_name] = result

        return self._results

    async def _check_one(self, source_name: str, adapter) -> dict:
        start = time.time()
        try:
            healthy = await asyncio.wait_for(
                adapter.health_check(), timeout=10.0
            )
            latency = int((time.time() - start) * 1000)
            logger.info(
                f"健康检查 [{source_name}]: "
                f"{'✅' if healthy else '❌'} {latency}ms"
            )
            return {
                "healthy":    healthy,
                "latency_ms": latency,
                "error":      "",
                "checked_at": time.strftime("%H:%M:%S"),
            }
        except asyncio.TimeoutError:
            return {
                "healthy":    False,
                "latency_ms": 10000,
                "error":      "健康检查超时",
                "checked_at": time.strftime("%H:%M:%S"),
            }
        except Exception as e:
            return {
                "healthy":    False,
                "latency_ms": -1,
                "error":      str(e),
                "checked_at": time.strftime("%H:%M:%S"),
            }

    def get_last_results(self) -> dict:
        return self._results
```
