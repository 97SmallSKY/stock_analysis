```python
# skills/stock_sim_robot/trade_cost.py
"""
交易成本计算模块

A股交易费用：
- 印花税：卖出 0.05%，买入免收
- 佣金：买卖双向，万3（0.03%），单笔最低5元
- 过户费：沪市 0.001%，深市免收
- 滑点：买入 +0.1%，卖出 -0.1%（模拟市场冲击）

所有费率可通过 CostConfig 自定义
"""
from dataclasses import dataclass, field


@dataclass
class CostConfig:
    """交易费率配置，可按机器人自定义"""
    stamp_tax_rate: float = 0.0005       # 印花税 0.05%
    commission_rate: float = 0.0003      # 佣金 万3
    commission_min: float = 5.0          # 佣金最低5元
    transfer_fee_rate_sh: float = 0.00001  # 沪市过户费 0.001%
    transfer_fee_rate_sz: float = 0.0      # 深市免收
    slippage_rate: float = 0.001         # 滑点 0.1%


# 全局默认配置
DEFAULT_COST_CONFIG = CostConfig()


def is_sh_stock(code: str) -> bool:
    """判断是否沪市股票：6开头"""
    return code.startswith(('6', '688'))


def calc_slippage_price(price: float, direction: str, config: CostConfig = None) -> float:
    """
    计算滑点后的实际成交价
    
    买入：价格上浮（模拟买入冲击）
    卖出：价格下浮（模拟卖出冲击）
    """
    if config is None:
        config = DEFAULT_COST_CONFIG
    
    if direction == 'BUY':
        return round(price * (1 + config.slippage_rate), 2)
    else:  # SELL
        return round(price * (1 - config.slippage_rate), 2)


def calc_buy_cost(price: float, shares: int, code: str, config: CostConfig = None) -> dict:
    """
    计算买入总成本
    
    Args:
        price: 成交价（已含滑点）
        shares: 股数
        code: 股票代码
        config: 费率配置
    
    Returns:
        {
            'amount': 成交金额,
            'commission': 佣金,
            'transfer_fee': 过户费,
            'stamp_tax': 0（买入免收）,
            'total_cost': 总费用,
            'total_paid': 实际支出（金额+费用）
        }
    """
    if config is None:
        config = DEFAULT_COST_CONFIG
    
    amount = price * shares
    
    # 佣金：万3，最低5元
    commission = max(amount * config.commission_rate, config.commission_min)
    
    # 过户费：沪市收，深市免
    if is_sh_stock(code):
        transfer_fee = amount * config.transfer_fee_rate_sh
    else:
        transfer_fee = 0.0
    
    # 印花税：买入免收
    stamp_tax = 0.0
    
    total_cost = round(commission + transfer_fee + stamp_tax, 2)
    total_paid = round(amount + total_cost, 2)
    
    return {
        'amount': round(amount, 2),
        'commission': round(commission, 2),
        'transfer_fee': round(transfer_fee, 2),
        'stamp_tax': stamp_tax,
        'slippage_cost': round(amount - price / (1 + config.slippage_rate) * shares, 2),
        'total_cost': total_cost,
        'total_paid': total_paid,
    }


def calc_sell_cost(price: float, shares: int, code: str, config: CostConfig = None) -> dict:
    """
    计算卖出总成本
    
    Returns:
        {
            'amount': 成交金额,
            'commission': 佣金,
            'transfer_fee': 过户费,
            'stamp_tax': 印花税,
            'total_cost': 总费用,
            'total_received': 实际到账（金额-费用）
        }
    """
    if config is None:
        config = DEFAULT_COST_CONFIG
    
    amount = price * shares
    
    # 佣金
    commission = max(amount * config.commission_rate, config.commission_min)
    
    # 过户费
    if is_sh_stock(code):
        transfer_fee = amount * config.transfer_fee_rate_sh
    else:
        transfer_fee = 0.0
    
    # 印花税：卖出收 0.05%
    stamp_tax = amount * config.stamp_tax_rate
    
    total_cost = round(commission + transfer_fee + stamp_tax, 2)
    total_received = round(amount - total_cost, 2)
    
    return {
        'amount': round(amount, 2),
        'commission': round(commission, 2),
        'transfer_fee': round(transfer_fee, 2),
        'stamp_tax': round(stamp_tax, 2),
        'slippage_cost': round(price * (1 - config.slippage_rate) * shares - amount, 2),
        'total_cost': total_cost,
        'total_received': total_received,
    }


def calc_trade_profit(
    buy_price: float, sell_price: float, 
    shares: int, code: str,
    config: CostConfig = None
) -> dict:
    """
    计算单笔交易完整盈亏（含所有成本）
    
    Returns:
        {
            'buy_cost': 买入成本明细,
            'sell_cost': 卖出成本明细,
            'gross_profit': 毛利润（不含费用）,
            'total_fees': 总费用（买+卖）,
            'net_profit': 净利润,
            'net_profit_pct': 净收益率,
        }
    """
    if config is None:
        config = DEFAULT_COST_CONFIG
    
    buy = calc_buy_cost(buy_price, shares, code, config)
    sell = calc_sell_cost(sell_price, shares, code, config)
    
    gross_profit = sell['amount'] - buy['amount']
    total_fees = buy['total_cost'] + sell['total_cost']
    net_profit = gross_profit - total_fees
    net_profit_pct = (net_profit / buy['total_paid']) * 100
    
    return {
        'buy_cost': buy,
        'sell_cost': sell,
        'gross_profit': round(gross_profit, 2),
        'total_fees': round(total_fees, 2),
        'net_profit': round(net_profit, 2),
        'net_profit_pct': round(net_profit_pct, 4),
    }
```

### 二、Bug1 修复：仓位计算

```python
# skills/stock_sim_robot/position_calc.py
"""
仓位计算模块

职责：根据 Phase A 的决策金额计算买入股数
替代原来 trade_runner.py 中的硬编码 BUY_AMOUNT
"""
import logging
from .trade_cost import calc_slippage_price, calc_buy_cost, DEFAULT_COST_CONFIG

logger = logging.getLogger(__name__)

# 仓位控制常量
DEFAULT_POSITION_RATIO = 0.1   # 默认仓位 10%
MAX_SINGLE_TRADE_RATIO = 0.3   # 单笔最大 30%


def calculate_buy_shares(
    planned_amount: float,
    raw_price: float,
    stock_code: str,
    total_capital: float = 100000,
) -> dict:
    """
    计算买入股数（含滑点和费用）
    
    Args:
        planned_amount: Phase A 决策的计划投入金额
        raw_price: 原始价格（未含滑点）
        stock_code: 股票代码
        total_capital: 机器人总资金
    
    Returns:
        {
            'shares': 买入股数,
            'slippage_price': 滑点后成交价,
            'planned_amount': 计划金额,
            'actual_amount': 实际成交金额,
            'cost_detail': 成本明细,
            'valid': 是否有效,
            'reason': 无效原因,
        }
    """
    result = {
        'shares': 0,
        'slippage_price': 0,
        'planned_amount': planned_amount,
        'actual_amount': 0,
        'cost_detail': {},
        'valid': False,
        'reason': '',
    }
    
    # Step 1: 校验 planned_amount
    if planned_amount is None or planned_amount <= 0:
        planned_amount = total_capital * DEFAULT_POSITION_RATIO
        logger.warning(
            f"buy_amount 无效，使用默认仓位: {planned_amount:.0f}"
        )
        result['planned_amount'] = planned_amount
    
    # Step 2: 单笔上限
    max_amount = total_capital * MAX_SINGLE_TRADE_RATIO
    if planned_amount > max_amount:
        logger.warning(
            f"buy_amount {planned_amount:.0f} 超过单笔上限 "
            f"{max_amount:.0f}，截断"
        )
        planned_amount = max_amount
        result['planned_amount'] = planned_amount
    
    # Step 3: 计算滑点后价格
    slippage_price = calc_slippage_price(raw_price, 'BUY')
    result['slippage_price'] = slippage_price
    
    # Step 4: 预估费用，倒推可用于买股票的金额
    # 总支出 = 股票金额 + 佣金 + 过户费
    # 佣金 ≈ 金额 * 0.03%，过户费 ≈ 金额 * 0.001%
    # 可买金额 ≈ planned_amount / (1 + 0.0003 + 0.00001)
    fee_ratio = DEFAULT_COST_CONFIG.commission_rate
    if stock_code.startswith(('6', '688')):
        fee_ratio += DEFAULT_COST_CONFIG.transfer_fee_rate_sh
    available_for_stock = planned_amount / (1 + fee_ratio)
    
    # Step 5: 计算股数（向下取整到100）
    raw_shares = available_for_stock / slippage_price
    shares = int(raw_shares / 100) * 100
    
    # Step 6: 至少100股
    if shares < 100:
        min_cost = calc_buy_cost(slippage_price, 100, stock_code)
        if planned_amount >= min_cost['total_paid']:
            shares = 100
        else:
            result['reason'] = (
                f"资金不足: 需要{min_cost['total_paid']:.0f},"
                f"可用{planned_amount:.0f}"
            )
            logger.warning(f"{stock_code} {result['reason']}")
            return result
    
    # Step 7: 计算实际成本
    cost = calc_buy_cost(slippage_price, shares, stock_code)
    
    # Step 8: 如果总支出超过计划金额，减100股
    while cost['total_paid'] > planned_amount and shares >= 200:
        shares -= 100
        cost = calc_buy_cost(slippage_price, shares, stock_code)
    
    if cost['total_paid'] > planned_amount and shares < 200:
        result['reason'] = f"调整后仍超出预算"
        logger.warning(f"{stock_code} {result['reason']}")
        return result
    
    # Step 9: 填充结果
    result['shares'] = shares
    result['actual_amount'] = cost['amount']
    result['cost_detail'] = cost
    result['valid'] = True
    
    usage = cost['total_paid'] / planned_amount * 100
    logger.info(
        f"📊 {stock_code} 仓位计算: "
        f"计划={planned_amount:.0f}, "
        f"价格={raw_price}→{slippage_price}(含滑点), "
        f"股数={shares}, "
        f"实际={cost['total_paid']:.0f}(含费用), "
        f"使用率={usage:.1f}%"
    )
    
    return result
```

```python
def add_column_if_missing(conn, inspector, table: str, column: str, col_type: str):
    """安全添加字段"""
    if table not in inspector.get_table_names():
        logger.warning(f"  表 {table} 不存在，跳过")
        return False
    
    existing = [c['name'] for c in inspector.get_columns(table)]

        return False
    
    conn.execute(text(f"ALTER TABLE {table} ADD COLUMN {column} {col_type}"))
```

```python
现金余额管理

职责：

class CashManager:

                (robot_id, tx_type, amount, balance_after,
                 stock_code, stock_name, trade_date, memo)

        资金转移（机器人继承时使用）
        从当前机器人转出，转入目标机器人

            balance = self._get_balance_in_tx(conn)

                VALUES (:rid, 'TRANSFER_OUT', :amount, :balance, :date, :memo)

                'date': trade_date,

                WHERE robot_id = :rid ORDER BY id DESC LIMIT 1
            '''), {'rid': target_robot_id}).fetchone()

            target_new_balance = target_balance + amount
            
            conn.execute(text('''
                INSERT INTO cash_ledger
                (robot_id, tx_type, amount, balance_after, trade_date, memo)
                VALUES (:rid, 'TRANSFER_IN', :amount, :balance, :date, :memo)

                'amount': amount,
                'balance': target_new_balance,
                'date': trade_date,
                'memo': f'继承自 {self.robot_id}',
            })
        
        logger.info(
            f"🔄 资金转移: {self.robot_id}(¥{new_balance:.2f}) "

            f"转移 ¥{amount:.2f}"
        )

            WHERE robot_id = :rid
            ORDER BY id DESC LIMIT 1
            FOR UPDATE
        '''), {'rid': self.robot_id}).fetchone()
        
        return float(result[0]) if result else 0.0
```

```python
# skills/stock_sim_robot/strategy_base.py
"""
策略基类 - 所有策略必须继承此类

新增策略步骤：
1. 在 modules/ 下创建 strategy_xxx.py
2. 继承 BaseStrategy 实现所有抽象方法
3. 在 strategy_registry 表中注册
"""
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional


@dataclass
class SelectionSignal:
    """选股信号"""
    stock_code: str
    stock_name: str
    score: float
    buy_amount: float        # 建议买入金额
    reason: str
    extra: dict = None       # 策略特有数据


@dataclass
class TradeSignal:
    """交易信号"""
    stock_code: str
    direction: str           # BUY / SELL
    price: float             # 建议价格
    shares: int              # 建议股数（SELL时可为0表示全部）
    reason: str
    urgency: str = 'normal'  # normal / urgent（止损用urgent）
    batch_index: int = 0     # 分批建仓批次号
    batch_total: int = 1     # 总批次数


class BaseStrategy(ABC):
    """策略基类"""
    
    def __init__(self, params: dict = None):
        self.params = params or {}
    
    @abstractmethod
    def get_name(self) -> str:
        """策略名称"""
        pass
    
    @abstractmethod
    def get_id(self) -> str:
        """策略ID"""
        pass
    
    @abstractmethod
    def select_stocks(self, trade_date: str, context: dict) -> list[SelectionSignal]:
        """
        选股（Phase A 调用）
        
        Args:
            trade_date: 交易日
            context: 上下文（持仓、资金、行情等）
        
        Returns:
            选股信号列表
        """
        pass
    
    @abstractmethod
    def should_buy(self, signal: SelectionSignal, realtime: dict, context: dict) -> Optional[TradeSignal]:

        判断是否买入（Phase B / 实时监控调用）
        
        Args:
            signal: 选股信号
            realtime: 实时行情
            context: 上下文
        
        Returns:
            买入信号 或 None
        """
        pass
    
    @abstractmethod
    def should_sell(self, position: dict, realtime: dict, context: dict) -> Optional[TradeSignal]:
        """
        判断是否卖出（实时监控调用）
        
        Args:

            卖出信号 或 None
```

```python
# skills/stock_sim_robot/strategy_loader.py
"""策略加载器 - 根据 strategy_id 动态加载策略类"""
import importlib
import logging
from utils.storage import storage
from sqlalchemy import text
from .strategy_base import BaseStrategy

logger = logging.getLogger(__name__)

# 策略实例缓存
_strategy_cache: dict[str, BaseStrategy] = {}


def load_strategy(strategy_id: str, params: dict = None) -> BaseStrategy:
    """
    根据 strategy_id 加载策略实例
    
    流程：
    1. 查 strategy_registry 表获取 module_path
    2. 动态 import 模块
    3. 调用模块中的 create_strategy(params) 工厂函数
    """
    cache_key = f"{strategy_id}:{hash(str(params))}"
    if cache_key in _strategy_cache:
        return _strategy_cache[cache_key]
    
    engine = storage._get_engine()
    with engine.connect() as conn:
        row = conn.execute(text('''
            SELECT module_path, default_params
            FROM strategy_registry
            WHERE strategy_id = :sid AND is_active = true
        '''), {'sid': strategy_id}).fetchone()
    
    if not row:
        raise ValueError(f"策略 {strategy_id} 未注册或已禁用")
    
    row = dict(row._mapping)
    module_path = row['module_path']
    default_params = row.get('default_params') or {}
    
    # 合并参数：默认参数 + 机器人自定义参数
    merged_params = {**default_params, **(params or {})}
    
    try:
        module = importlib.import_module(module_path)
        strategy = module.create_strategy(merged_params)
        
        if not isinstance(strategy, BaseStrategy):
            raise TypeError(
                f"{module_path}.create_strategy() 返回类型不是 BaseStrategy"
            )
        
        _strategy_cache[cache_key] = strategy
        logger.info(f"✅ 加载策略: {strategy.get_name()} ({strategy_id})")
        return strategy
        
    except Exception as e:
        logger.error(f"❌ 加载策略 {strategy_id} 失败: {e}")
        raise


def list_strategies() -> list[dict]:
    """列出所有已注册策略"""
    engine = storage._get_engine()
    with engine.connect() as conn:
        rows = conn.execute(text('''
            SELECT strategy_id, strategy_name, strategy_type,
                   description, support_batch, is_active
            FROM strategy_registry
            ORDER BY id
        ''')).fetchall()
    
    return [dict(r._mapping) for r in rows]
```

```python
# skills/stock_sim_robot/modules/strategy_lianban.py
"""
龙虎榜多因子策略

将现有的选股/买卖逻辑包装为 BaseStrategy 接口
后续新增策略照此模式创建新文件即可
"""
import logging
from ..strategy_base import BaseStrategy, SelectionSignal, TradeSignal
from typing import Optional

logger = logging.getLogger(__name__)


class LianbanMultiFactorStrategy(BaseStrategy):
    """龙虎榜多因子策略"""
    
    def get_name(self) -> str:
        return "龙虎榜多因子策略"
    
    def get_id(self) -> str:
        return "lianban_multi_factor"
    
    def select_stocks(self, trade_date: str, context: dict) -> list[SelectionSignal]:
        """
        选股逻辑（复用现有 Phase A 代码）
        
        context 包含：
        - robot: 机器人配置
        - holdings: 当前持仓列表
        - cash: 当前现金余额
        """
        robot = context['robot']
        cash = context['cash']
        holdings = context.get('holdings', [])
        max_positions = robot.get('max_positions', 3)
        
        # 可买入名额
        available_slots = max_positions - len(holdings)
        if available_slots <= 0:
            logger.info(f"持仓已满 ({len(holdings)}/{max_positions})，跳过选股")
            return []
        
        # 单笔金额
        position_ratio = robot.get('position_ratio', 0.1)
        buy_amount = robot['initial_capital'] * position_ratio
        
        # 调用现有选股逻辑
        # TODO: 将现有 llm_select.py 的核心逻辑抽取到这里
        # 临时方案：导入现有函数
        try:
            from ..llm_select import run_selection
            candidates = run_selection(trade_date, self.params)
        except ImportError:
            logger.warning("llm_select 模块不可用，返回空选股")
            return []
        
        signals = []
        for c in candidates[:available_slots]:
            signals.append(SelectionSignal(
                stock_code=c['code'],
                stock_name=c['name'],
                score=c.get('score', 0),
                buy_amount=min(buy_amount, cash),
                reason=c.get('reason', ''),
                extra={'lianban_score': c.get('lianban_score', 0)},
            ))
            cash -= buy_amount  # 预扣现金
        
        return signals
    
    def should_buy(self, signal: SelectionSignal, realtime: dict, context: dict) -> Optional[TradeSignal]:
        """
        买入判断
        
        默认逻辑：开盘后直接买入（T+1策略）
        """
        price = realtime.get('price') or realtime.get('open')
        if not price or price <= 0:
            return None
        
        # 涨停不买
        pre_close = realtime.get('pre_close', 0)
        if pre_close > 0:
            change_pct = (price - pre_close) / pre_close * 100
            if change_pct >= 9.8:
                logger.info(f"{signal.stock_code} 涨停({change_pct:.1f}%)，不追")
                return None
        
        return TradeSignal(
            stock_code=signal.stock_code,
            direction='BUY',
            price=price,
            shares=0,  # 由 position_calc 计算
            reason=f"T+1买入，选股评分{signal.score:.1f}",
        )
    
    def should_sell(self, position: dict, realtime: dict, context: dict) -> Optional[TradeSignal]:
        """
        卖出判断
        
        触发条件（按优先级）：
        1. 止损：跌破止损线
        2. 到期：超过最长持有天数
        3. 止盈：达到止盈线（如果设置了）
        """
        robot = context['robot']
        price = realtime.get('price', 0)
        buy_price = position.get('buy_price', 0)
        
        if not price or not buy_price:
            return None
        
        profit_pct = (price - buy_price) / buy_price
        
        # 止损检查
        stop_loss = robot.get('stop_loss_pct', -0.5)
        if profit_pct <= stop_loss:
            return TradeSignal(
                stock_code=position['stock_code'],
                direction='SELL',
                price=price,
                shares=0,
                reason=f"止损触发: {profit_pct:.1%} <= {stop_loss:.1%}",
                urgency='urgent',
            )
        
        # 到期检查
        from datetime import datetime
        buy_date = position.get('trade_date', '')
        if buy_date:
            try:
                days_held = (datetime.now() - datetime.strptime(buy_date, '%Y%m%d')).days
                max_days = robot.get('max_holding_days', 7)
                if days_held >= max_days:
                    return TradeSignal(
                        stock_code=position['stock_code'],
                        direction='SELL',
                        price=price,
                        shares=0,
                        reason=f"持仓到期: {days_held}天 >= {max_days}天",
                    )
            except ValueError:
                pass
        
        # 止盈检查
        stop_profit = robot.get('stop_profit_pct')
        if stop_profit and profit_pct >= stop_profit:
            return TradeSignal(
                stock_code=position['stock_code'],
                direction='SELL',
                price=price,
                shares=0,
                reason=f"止盈触发: {profit_pct:.1%} >= {stop_profit:.1%}",
            )
        
        return None


def create_strategy(params: dict) -> BaseStrategy:
    """工厂函数（策略加载器调用）"""
    return LianbanMultiFactorStrategy(params)
```

```python
            if robot['status'] not in ('draft', 'paused'):
                raise ValueError(
                    f"机器人 {robot_id} 状态为 {robot['status']}，无法启动"
                )
            
            conn.execute(text('''
                UPDATE sim_robots 
                SET status = 'running', started_at = NOW(), updated_at = NOW()
                WHERE robot_id = :rid
            '''), {'rid': robot_id})
        
        # 初始化现金（仅首次启动）
        cm = CashManager(robot_id)
        if cm.get_balance() == 0:
            cm.init_capital(robot['initial_capital'], trade_date)
        
        logger.info(f"🟢 机器人已启动: {robot_id}")
    
    def pause_robot(self, robot_id: str):
        """暂停机器人（running → paused）"""
        with self.engine.begin() as conn:
            conn.execute(text('''
                UPDATE sim_robots 
                SET status = 'paused', updated_at = NOW()
                WHERE robot_id = :rid AND status = 'running'
            '''), {'rid': robot_id})
        
        logger.info(f"⏸️ 机器人已暂停: {robot_id}")
    
    def end_robot(self, robot_id: str, trade_date: str = None):
        """
        结束机器人（强制清仓 → ended）
        """
        if trade_date is None:
            trade_date = datetime.now().strftime('%Y%m%d')
        
        with self.engine.begin() as conn:
            # 强制清仓
            holdings = conn.execute(text('''
                SELECT id, stock_code, stock_name, buy_price, buy_shares
                FROM sim_trades
                WHERE robot_id = :rid AND status = 'HOLDING'
            '''), {'rid': robot_id}).fetchall()
            
            if holdings:
                logger.info(f"清仓 {len(holdings)} 只持仓...")
                for h in holdings:
                    h = dict(h._mapping)
                    # 获取收盘价
                    from ..eastmoney_api import EastMoneyAPI
                    api = EastMoneyAPI()
                    quote = api.realtime(h['stock_code'])
                    close_price = quote.get('price') or h['buy_price']
                    
                    # 调用卖出
                    from .trade_runner import execute_sell
                    execute_sell(
                        pos=h, raw_price=close_price,
                        trade_date=trade_date, conn=conn,
                        reason='机器人结束-强制清仓'
                    )
            
            # 更新状态
            conn.execute(text('''
                UPDATE sim_robots 
                SET status = 'ended', ended_at = NOW(), updated_at = NOW()
                WHERE robot_id = :rid
            '''), {'rid': robot_id})
        
        logger.info(f"🔴 机器人已结束: {robot_id}，清仓 {len(holdings)} 只")
    
    def fork_robot(self, source_robot_id: str, new_config: dict, trade_date: str = None) -> str:
        """
        修改参数 = 创建新机器人（继承持仓和对应现金）
        
        流程：
        1. 创建新机器人（用新参数）
        2. 复制当前持仓到新机器人
        3. 计算持仓占用资金，从原机器人转移对应现金
        4. 原机器人继续运行不受影响
        
        Args:
            source_robot_id: 原机器人ID
            new_config: 新参数（会覆盖原参数）
        
        Returns:
            新机器人 robot_id
        """
        if trade_date is None:
            trade_date = datetime.now().strftime('%Y%m%d')
        
        with self.engine.begin() as conn:
            # 读取原机器人配置
            source = self._get_robot(conn, source_robot_id)
            
            if source['status'] != 'running':
                raise ValueError(f"只能从运行中的机器人继承，当前状态: {source['status']}")
            
            # 合并配置：原配置 + 新配置
            merged = {
                'robot_name': new_config.get('robot_name', f"{source['robot_name']}-分支"),
                'stock_source': new_config.get('stock_source', source['stock_source']),
                'select_strategy': new_config.get('select_strategy', source['select_strategy']),
                'execute_strategy': new_config.get('execute_strategy', source['execute_strategy']),
                'initial_capital': source['initial_capital'],  # 初始资金不变
                'position_ratio': new_config.get('position_ratio', source['position_ratio']),
                'max_positions': new_config.get('max_positions', source['max_positions']),
                'max_holding_days': new_config.get('max_holding_days', source['max_holding_days']),
                'stop_loss_pct': new_config.get('stop_loss_pct', source['stop_loss_pct']),
                'stop_profit_pct': new_config.get('stop_profit_pct', source.get('stop_profit_pct')),
                'enable_batch_buy': new_config.get('enable_batch_buy', source['enable_batch_buy']),
                'strategy_params': new_config.get('strategy_params', source.get('strategy_params', {})),
                'cost_config': new_config.get('cost_config', source.get('cost_config', {})),
            }
        
        # 创建新机器人
        new_robot_id = self.create_robot(merged)
        
        with self.engine.begin() as conn:
            # 记录继承关系
            conn.execute(text('''
                UPDATE sim_robots 
                SET parent_robot_id = :parent
                WHERE robot_id = :rid
            '''), {'parent': source_robot_id, 'rid': new_robot_id})
            
            # 复制持仓
            holdings = conn.execute(text('''
                SELECT stock_code, stock_name, trade_date, buy_amount,
                       buy_price, buy_shares, buy_slippage_price,
                       buy_commission, buy_transfer_fee
                FROM sim_trades
                WHERE robot_id = :rid AND status = 'HOLDING'
            '''), {'rid': source_robot_id}).fetchall()
            
            holding_cost = 0.0
            for h in holdings:
                h = dict(h._mapping)
                cost = h['buy_price'] * h['buy_shares']
                holding_cost += cost
                
                # 写入新机器人的持仓记录
                conn.execute(text('''
                    INSERT INTO sim_trades
                    (robot_id, stock_code, stock_name, trade_date,
                     buy_amount, buy_price, buy_shares, buy_slippage_price,
                     buy_commission, buy_transfer_fee, status, created_at)
                    VALUES
                    (:rid, :code, :name, :date,
                     :amount, :price, :shares, :slippage,
                     :commission, :transfer, 'HOLDING', NOW())
                '''), {
                    'rid': new_robot_id,
                    'code': h['stock_code'],
                    'name': h['stock_name'],
                    'date': h['trade_date'],
                    'amount': h['buy_amount'],
                    'price': h['buy_price'],
                    'shares': h['buy_shares'],
                    'slippage': h.get('buy_slippage_price'),
                    'commission': h.get('buy_commission', 0),
                    'transfer': h.get('buy_transfer_fee', 0),
                })
            
            logger.info(f"📋 复制 {len(holdings)} 只持仓，占用 ¥{holding_cost:.2f}")
        
        # 资金处理
        source_cm = CashManager(source_robot_id)
        source_balance = source_cm.get_balance()
        
        # 判断是否修改了仓位相关参数
        ratio_changed = new_config.get('position_ratio') and \
                        new_config['position_ratio'] != source['position_ratio']
        
        if ratio_changed:
            # 仓位比例变了，按新比例重新分配现金
            # 新机器人获得：持仓占用 + 按比例的剩余现金
            transfer_cash = source_balance * (new_config['position_ratio'] / source['position_ratio'])
            transfer_cash = min(transfer_cash, source_balance)
        else:
            # 参数未涉及资金，现金等额复制
            transfer_cash = source_balance
        
        # 初始化新机器人现金
        new_cm = CashManager(new_robot_id)
        new_cm.init_capital(0, trade_date)  # 先初始化为0
        source_cm.transfer(new_robot_id, transfer_cash, trade_date)
        
        # 启动新机器人
        self.start_robot(new_robot_id, trade_date)
        
        logger.info(
            f"🔀 机器人继承完成: {source_robot_id} → {new_robot_id}, "
            f"持仓 {len(holdings)} 只, 现金 ¥{transfer_cash:.2f}"
        )
        
        return new_robot_id
    
    def list_robots(self, status: str = None) -> list[dict]:
        """列出机器人"""
        with self.engine.connect() as conn:
            if status:
                rows = conn.execute(text('''
                    SELECT r.*, 
                           (SELECT COUNT(*) FROM sim_trades 
                            WHERE robot_id = r.robot_id AND status = 'HOLDING') as holding_count,
                           (SELECT balance_after FROM cash_ledger 
                            WHERE robot_id = r.robot_id 
                            ORDER BY id DESC LIMIT 1) as cash_balance
                    FROM sim_robots r
                    WHERE r.status = :status
                    ORDER BY r.created_at DESC
                '''), {'status': status}).fetchall()
            else:
                rows = conn.execute(text('''
                    SELECT r.*,
                           (SELECT COUNT(*) FROM sim_trades 
                            WHERE robot_id = r.robot_id AND status = 'HOLDING') as holding_count,
                           (SELECT balance_after FROM cash_ledger 
                            WHERE robot_id = r.robot_id 
                            ORDER BY id DESC LIMIT 1) as cash_balance
                    FROM sim_robots r
                    ORDER BY r.created_at DESC
                ''')).fetchall()
        
        return [dict(r._mapping) for r in rows]
    
    def get_robot_detail(self, robot_id: str) -> dict:
        """获取单个机器人详情"""
        with self.engine.connect() as conn:
            robot = self._get_robot(conn, robot_id)
            
            # 持仓
            holdings = conn.execute(text('''
                SELECT * FROM sim_trades
                WHERE robot_id = :rid AND status = 'HOLDING'
                ORDER BY trade_date DESC
            '''), {'rid': robot_id}).fetchall()
            
            # 历史交易
            history = conn.execute(text('''
                SELECT * FROM sim_trades
                WHERE robot_id = :rid AND status = 'SOLD'
                ORDER BY sell_date DESC
            '''), {'rid': robot_id}).fetchall()
            
            # 现金余额
            cash = CashManager(robot_id).get_balance()
            
            # 持仓市值（用最新价估算）
            holding_value = 0
            for h in holdings:
                h = dict(h._mapping)
                holding_value += h['buy_price'] * h['buy_shares']  # 简化，P4会用实时价
            
            return {
                'robot': robot,
                'holdings': [dict(h._mapping) for h in holdings],
                'history': [dict(h._mapping) for h in history],
                'cash': cash,
                'holding_value': holding_value,
                'total_value': cash + holding_value,
                'nav': (cash + holding_value) / robot['initial_capital'],
            }
    
    def _get_robot(self, conn, robot_id: str) -> dict:
        """查询单个机器人"""
        row = conn.execute(text(
            'SELECT * FROM sim_robots WHERE robot_id = :rid'
        ), {'rid': robot_id}).fetchone()
        
        if not row:
            raise ValueError(f"机器人 {robot_id} 不存在")
        
        return dict(row._mapping)
```

### 二、每日结算任务

```python
# skills/stock_sim_robot/daily_settlement.py
"""
每日结算任务（15:05 执行）

职责：
1. 获取所有持仓股票的收盘价
2. 计算每个机器人的持仓市值和净值
3. 检查超期持仓强制卖出
4. 写入 sim_daily_summary
5. 拉取基准指数收盘数据
"""
import logging
from datetime import datetime
from utils.storage import storage
from sqlalchemy import text
from .robot_manager import RobotManager
from .cash_manager import CashManager

logger = logging.getLogger(__name__)


class DailySettlement:
    
    def __init__(self):
        self.engine = storage._get_engine()
        self.rm = RobotManager()
    
    def run(self, trade_date: str = None):
        """执行每日结算"""
        if trade_date is None:
            trade_date = datetime.now().strftime('%Y%m%d')
        
        logger.info(f"=== 每日结算开始: {trade_date} ===")
        
        robots = self.rm.list_robots(status='running')
        
        # 批量获取所有持仓股票的收盘价
        close_prices = self._fetch_close_prices(robots)
        
        for robot in robots:
            robot_id = robot['robot_id']
            try:
                self._settle_one_robot(robot, trade_date, close_prices)
            except Exception as e:
                logger.error(f"❌ {robot_id} 结算异常: {e}", exc_info=True)
        
        # 拉取基准指数
        self._fetch_benchmarks(trade_date)
        
        logger.info(f"=== 每日结算完成 ===")
    
    def _fetch_close_prices(self, robots: list) -> dict:
        """批量获取收盘价"""
        from skills.eastmoney_api import EastMoneyAPI
        api = EastMoneyAPI()
        
        # 收集所有持仓股票代码
        all_codes = set()
        with self.engine.connect() as conn:
            for robot in robots:
                holdings = conn.execute(text('''
                    SELECT stock_code FROM sim_trades
                    WHERE robot_id = :rid AND status = 'HOLDING'
                '''), {'rid': robot['robot_id']}).fetchall()
                
                for h in holdings:
                    all_codes.add(h[0])
        
        if not all_codes:
            return {}
        
        # 批量获取
        prices = {}
        for code in all_codes:
            try:
                quote = api.realtime(code)
                prices[code] = {
                    'price': quote.get('price', 0),
                    'pre_close': quote.get('pre_close', 0),
                    'change_pct': quote.get('change_pct', 0),
                }
            except Exception as e:
                logger.warning(f"获取 {code} 行情失败: {e}")
        
        logger.info(f"📊 获取 {len(prices)}/{len(all_codes)} 只股票收盘价")
        return prices
    
    def _settle_one_robot(self, robot: dict, trade_date: str, close_prices: dict):
        """结算单个机器人"""
        robot_id = robot['robot_id']
        cm = CashManager(robot_id)
        cash = cm.get_balance()
        
        with self.engine.begin() as conn:
            # 查询持仓
            holdings = conn.execute(text('''
                SELECT id, stock_code, stock_name, buy_price, buy_shares, trade_date
                FROM sim_trades
                WHERE robot_id = :rid AND status = 'HOLDING'
            '''), {'rid': robot_id}).fetchall()
            
            holding_value = 0.0
            holdings_snapshot = []
            
            for h in holdings:
                h = dict(h._mapping)
                code = h['stock_code']
                
                # 获取收盘价
                quote = close_prices.get(code, {})
                current_price = quote.get('price') or h['buy_price']
                
                market_value = current_price * h['buy_shares']
                holding_value += market_value
                
                profit_pct = (current_price - h['buy_price']) / h['buy_price'] * 100
                
                holdings_snapshot.append({
                    'code': code,
                    'name': h['stock_name'],
                    'shares': h['buy_shares'],
                    'buy_price': h['buy_price'],
                    'current_price': current_price,
                    'market_value': round(market_value, 2),
                    'profit_pct': round(profit_pct, 2),
                })
                
                # 检查超期
                self._check_expiry(h, robot, trade_date, current_price, conn)
            
            # 计算净值
            total_value = cash + holding_value
            initial = robot['initial_capital']
            nav = total_value / initial
            
            # 查昨日净值算日收益
            prev = conn.execute(text('''
                SELECT nav, total_value FROM sim_daily_summary
                WHERE robot_id = :rid AND trade_date < :date
                ORDER BY trade_date DESC LIMIT 1
            '''), {'rid': robot_id, 'date': trade_date}).fetchone()
            
            if prev:
                prev_nav = prev[0]
                prev_value = prev[1]
                daily_return = total_value - prev_value
                daily_return_pct = (nav / prev_nav - 1) * 100
            else:
                daily_return = total_value - initial
                daily_return_pct = (nav - 1) * 100
            
            # 累计收益
            cumulative_return_pct = (nav - 1) * 100
            
            # 最大回撤
            peak_nav = conn.execute(text('''
                SELECT COALESCE(MAX(nav), 1.0) FROM sim_daily_summary
                WHERE robot_id = :rid
            '''), {'rid': robot_id}).scalar()
            
            peak_nav = max(float(peak_nav), nav)
            max_dd = (nav - peak_nav) / peak_nav * 100 if peak_nav > 0 else 0
            
            # 累计费用
            total_fees = conn.execute(text('''
                SELECT COALESCE(SUM(total_fees), 0)
                FROM sim_trades
                WHERE robot_id = :rid AND total_fees IS NOT NULL
            '''), {'rid': robot_id}).scalar()
            
            # 写入快照
            import json
            conn.execute(text('''
                INSERT INTO sim_daily_summary
                (robot_id, trade_date, cash, holding_value, total_value, nav,
                 daily_return, daily_return_pct, cumulative_return_pct,
                 max_drawdown_pct, holdings_snapshot, total_fees_cumulative)
                VALUES
                (:rid, :date, :cash, :hv, :tv, :nav,
                 :dr, :drp, :crp, :mdd, :snap::jsonb, :fees)
                ON CONFLICT (robot_id, trade_date) 
                DO UPDATE SET 
                    cash = EXCLUDED.cash,
                    holding_value = EXCLUDED.holding_value,
                    total_value = EXCLUDED.total_value,
                    nav = EXCLUDED.nav,
                    daily_return = EXCLUDED.daily_return,
                    daily_return_pct = EXCLUDED.daily_return_pct,
                    cumulative_return_pct = EXCLUDED.cumulative_return_pct,
                    max_drawdown_pct = EXCLUDED.max_drawdown_pct,
                    holdings_snapshot = EXCLUDED.holdings_snapshot,
                    total_fees_cumulative = EXCLUDED.total_fees_cumulative
            '''), {
                'rid': robot_id, 'date': trade_date,
                'cash': round(cash, 2),
                'hv': round(holding_value, 2),
                'tv': round(total_value, 2),
                'nav': round(nav, 6),
                'dr': round(daily_return, 2),
                'drp': round(daily_return_pct, 4),
                'crp': round(cumulative_return_pct, 4),
                'mdd': round(max_dd, 4),
                'snap': json.dumps(holdings_snapshot, ensure_ascii=False),
                'fees': round(float(total_fees), 2),
            })
        
        logger.info(
            f"📊 {robot['robot_name']}: "
            f"现金=¥{cash:.0f}, 持仓=¥{holding_value:.0f}, "
            f"总值=¥{total_value:.0f}, 净值={nav:.4f}, "
            f"日收益={daily_return_pct:+.2f}%"
        )
    
    def _check_expiry(self, holding: dict, robot: dict, trade_date: str, 
                      current_price: float, conn):
        """检查持仓是否超期，超期则强制卖出"""
        max_days = robot.get('max_holding_days', 7)
        buy_date = holding.get('trade_date', '')
        
        if not buy_date:
            return
        
        try:
            days_held = (
                datetime.strptime(trade_date, '%Y%m%d') - 
                datetime.strptime(buy_date, '%Y%m%d')
            ).days
        except ValueError:
            return
        
        if days_held >= max_days:
            logger.info(
                f"⏰ {holding['stock_code']} 持仓 {days_held} 天，"
                f"超过上限 {max_days} 天，强制卖出"
            )
            execute_sell(
                pos=holding, raw_price=current_price,
                trade_date=trade_date, conn=conn,
                reason=f'持仓到期({days_held}/{max_days}天)'
            )
            
            # 卖出到账
            from .trade_cost import calc_sell_cost, calc_slippage_price
            sell_price = calc_slippage_price(current_price, 'SELL')
            sell_cost = calc_sell_cost(sell_price, holding['buy_shares'], holding['stock_code'])
            
            cm = CashManager(robot['robot_id'])
            cm.credit_sell(
                total_received=sell_cost['total_received'],
                trade_id=holding['id'],
                stock_code=holding['stock_code'],
                stock_name=holding['stock_name'],
                trade_date=trade_date,
            )
    
    def _fetch_benchmarks(self, trade_date: str):
        """拉取基准指数数据"""
        from skills.eastmoney_api import EastMoneyAPI
        api = EastMoneyAPI()
        
        benchmarks = {
            '000300': '沪深300',
            '000905': '中证500',
            '000001': '上证指数',
            '399006': '创业板指',
        }
        
        with self.engine.begin() as conn:
            for code, name in benchmarks.items():
                try:
                    # 获取指数行情
                    quote = api.realtime(code)
                    price = quote.get('price', 0)
                    change_pct = quote.get('change_pct', 0)
                    
                    if not price:
                        continue
                    
                    # 计算相对净值（以首日为基准1.0）
                    first = conn.execute(text('''
                        SELECT close_price FROM benchmark_daily
                        WHERE index_code = :code
                        ORDER BY trade_date ASC LIMIT 1
                    '''), {'code': code}).fetchone()
                    
                    base_price = first[0] if first else price
                    nav = price / base_price
                    
                    conn.execute(text('''
                        INSERT INTO benchmark_daily 
                        (index_code, index_name, trade_date, close_price, change_pct, nav)
                        VALUES (:code, :name, :date, :price, :pct, :nav)
                        ON CONFLICT (index_code, trade_date)
                        DO UPDATE SET close_price = EXCLUDED.close_price,
                                      change_pct = EXCLUDED.change_pct,
                                      nav = EXCLUDED.nav
                    '''), {
                        'code': code, 'name': name, 'date': trade_date,
                        'price': price, 'pct': change_pct, 'nav': round(nav, 6),
                    })
                    
                except Exception as e:
                    logger.warning(f"获取基准 {code} 失败: {e}")
        
        logger.info(f"📈 基准指数数据已更新")
```

```python
# skills/stock_sim_robot/scorer.py
"""
机器人评分体系

默认权重（页面可调）：
- 累计收益率    25%
- 最大回撤      20%
- 夏普比率      20%
- 胜率          10%
- 盈亏比        10%
- 交易成本占比  10%
- 样本数量       5%
"""
import logging
import math
from utils.storage import storage
from sqlalchemy import text

logger = logging.getLogger(__name__)

DEFAULT_WEIGHTS = {
    'return_rate': 0.25,
    'max_drawdown': 0.20,
    'sharpe_ratio': 0.20,
    'win_rate': 0.10,
    'profit_loss_ratio': 0.10,
    'cost_ratio': 0.10,
    'sample_size': 0.05,
}

# 评分标准：每个指标的满分参考值
SCORING_BENCHMARKS = {
    'return_rate': {'perfect': 50, 'zero': 0},         # 50%收益=满分
    'max_drawdown': {'perfect': 0, 'zero': -30},       # 0回撤=满分，-30%=0分
    'sharpe_ratio': {'perfect': 3.0, 'zero': 0},       # 夏普3=满分
    'win_rate': {'perfect': 80, 'zero': 30},            # 80%胜率=满分
    'profit_loss_ratio': {'perfect': 3.0, 'zero': 0.5}, # 盈亏比3:1=满分
    'cost_ratio': {'perfect': 0, 'zero': 5},            # 成本占比0%=满分
    'sample_size': {'perfect': 30, 'zero': 3},          # 30笔交易=满分
}


def calc_robot_metrics(robot_id: str) -> dict:
    """计算单个机器人的所有指标"""
    engine = storage._get_engine()
    
    with engine.connect() as conn:
        # 累计收益率
        nav_rows = conn.execute(text('''
            SELECT trade_date, nav, daily_return_pct, total_fees_cumulative
            FROM sim_daily_summary
            WHERE robot_id = :rid
            ORDER BY trade_date
        '''), {'rid': robot_id}).fetchall()
        
        if not nav_rows:
            return None
        
        navs = [dict(r._mapping) for r in nav_rows]
        latest = navs[-1]
        return_rate = (latest['nav'] - 1) * 100
        
        # 最大回撤
        peak = 1.0
        max_dd = 0.0
        for n in navs:
            peak = max(peak, n['nav'])
            dd = (n['nav'] - peak) / peak * 100
            max_dd = min(max_dd, dd)
        
        # 夏普比率（年化，无风险利率2%）
        daily_returns = [n['daily_return_pct'] for n in navs if n['daily_return_pct'] is not None]
        if len(daily_returns) > 1:
            avg_daily = sum(daily_returns) / len(daily_returns)
            std_daily = math.sqrt(sum((r - avg_daily) ** 2 for r in daily_returns) / (len(daily_returns) - 1))
            risk_free_daily = 2.0 / 252  # 年化2%转日
            sharpe = (avg_daily - risk_free_daily) / std_daily * math.sqrt(252) if std_daily > 0 else 0
        else:
            sharpe = 0
        
        # 胜率和盈亏比
        trades = conn.execute(text('''
            SELECT net_profit, net_profit_pct
            FROM sim_trades
            WHERE robot_id = :rid AND status = 'SOLD' AND net_profit IS NOT NULL
        '''), {'rid': robot_id}).fetchall()
        
        total_trades = len(trades)
        if total_trades > 0:
            wins = [t for t in trades if t[0] > 0]
            losses = [t for t in trades if t[0] <= 0]
            win_rate = len(wins) / total_trades * 100
            
            avg_win = sum(t[1] for t in wins) / len(wins) if wins else 0
            avg_loss = abs(sum(t[1] for t in losses) / len(losses)) if losses else 1
            profit_loss_ratio = avg_win / avg_loss if avg_loss > 0 else 0
        else:
            win_rate = 0
            profit_loss_ratio = 0
        
        # 交易成本占比
        total_fees = float(latest.get('total_fees_cumulative', 0))
        initial = conn.execute(text(
            'SELECT initial_capital FROM sim_robots WHERE robot_id = :rid'
        ), {'rid': robot_id}).scalar() or 100000
        cost_ratio = total_fees / float(initial) * 100
        
        return {
            'return_rate': round(return_rate, 2),
            'max_drawdown': round(max_dd, 2),
            'sharpe_ratio': round(sharpe, 2),
            'win_rate': round(win_rate, 2),
            'profit_loss_ratio': round(profit_loss_ratio, 2),
            'cost_ratio': round(cost_ratio, 2),
            'sample_size': total_trades,
            'total_days': len(navs),
            'latest_nav': round(latest['nav'], 4),
        }


def normalize_score(value: float, metric: str) -> float:
    """将原始指标归一化到 0-100 分"""
    bench = SCORING_BENCHMARKS.get(metric)
    if not bench:
        return 50
    
    perfect = bench['perfect']
    zero = bench['zero']
    
    if perfect == zero:
        return 50
    
    # 线性映射
    score = (value - zero) / (perfect - zero) * 100
    return max(0, min(100, score))


def calc_robot_score(robot_id: str, weights: dict = None) -> dict:
    """
    计算机器人综合评分
    
    Returns:
        {
            'total_score': 综合评分 0-100,
            'metrics': 各指标原始值,
            'scores': 各指标得分,
            'weighted_scores': 各指标加权得分,
            'weights': 使用的权重,
        }
    """
    if weights is None:
        weights = DEFAULT_WEIGHTS
    
    metrics = calc_robot_metrics(robot_id)
    if not metrics:
        return None
    
    scores = {}
    weighted_scores = {}
    
    for metric, weight in weights.items():
        raw_value = metrics.get(metric, 0)
        score = normalize_score(raw_value, metric)
        scores[metric] = round(score, 1)
        weighted_scores[metric] = round(score * weight, 2)
    
    total_score = sum(weighted_scores.values())
    
    return {
        'total_score': round(total_score, 1),
        'metrics': metrics,
        'scores': scores,
        'weighted_scores': weighted_scores,
        'weights': weights,
    }


def rank_robots(robot_ids: list, weights: dict = None) -> list[dict]:
    """多机器人排名"""
    results = []
    for rid in robot_ids:
        score = calc_robot_score(rid, weights)
        if score:
            score['robot_id'] = rid
            results.append(score)
    
    results.sort(key=lambda x: x['total_score'], reverse=True)
    
    for i, r in enumerate(results):
        r['rank'] = i + 1
    
    return results
```

```python
def chart_monthly_heatmap(robot_id: str) -> go.Figure:
    """单机器人月度收益热力图"""
    engine = storage._get_engine()
    
    with engine.connect() as conn:
        name = conn.execute(text(
            'SELECT robot_name FROM sim_robots WHERE robot_id = :rid'
        ), {'rid': robot_id}).scalar() or robot_id
        
        df = pd.read_sql(text('''
            SELECT trade_date, nav FROM sim_daily_summary
            WHERE robot_id = :rid ORDER BY trade_date
        '''), conn, params={'rid': robot_id})
    
    if df.empty:
        return go.Figure().update_layout(title=f"{name}: 无数据")
    
    df['date'] = pd.to_datetime(df['trade_date'], format='%Y%m%d')
    df['year'] = df['date'].dt.year
    df['month'] = df['date'].dt.month
    
    # 每月首尾净值算月收益
    monthly = df.groupby(['year', 'month']).agg(
        first_nav=('nav', 'first'),
        last_nav=('nav', 'last'),
    ).reset_index()
    monthly['return_pct'] = (monthly['last_nav'] / monthly['first_nav'] - 1) * 100
    
    pivot = monthly.pivot(index='year', columns='month', values='return_pct')
    month_labels = [f'{m}月' for m in range(1, 13)]
    
    # 补齐12个月
    for m in range(1, 13):
        if m not in pivot.columns:
            pivot[m] = None
    pivot = pivot[sorted(pivot.columns)]
    
    fig = go.Figure(data=go.Heatmap(
        z=pivot.values,
        x=month_labels[:len(pivot.columns)],
        y=[str(y) for y in pivot.index],
        colorscale='RdYlGn',
        zmid=0,
        text=[[f'{v:.1f}%' if pd.notna(v) else '' for v in row] for row in pivot.values],
        texttemplate='%{text}',
        hovertemplate='%{y}年%{x}: %{z:.2f}%<extra></extra>',
        colorbar_title='收益率(%)',
    ))
    
    fig.update_layout(
        title=f"📅 {name} 月度收益热力图",
        height=300,
        template='plotly_white',
    )
    
    return fig

                    "收益率: %{y:.2f}%<br>"
```

### 二、实时监控主循环

```python
# skills/stock_sim_robot/realtime_monitor.py
"""
实时监控服务

运行时段：交易日 9:30 - 15:00
轮询频率：每10秒获取实时行情
存储频率：每5分钟写入一条快照
触发动作：止损/止盈立即执行模拟卖出

与 Phase A/B 的关系：
- Phase A (18:45): 盘后选股，不冲突
- Phase B (09:35): 开盘买入，监控从09:36开始避免冲突
- 监控 (09:36-15:00): 盘中持续运行
"""
import time
import logging
from datetime import datetime, timedelta
from utils.storage import storage
from sqlalchemy import text
from .robot_manager import RobotManager
from .cash_manager import CashManager
from .strategy_loader import load_strategy
from .trade_cost import calc_slippage_price, calc_sell_cost
from .trade_runner import execute_sell

logger = logging.getLogger(__name__)

# 配置
POLL_INTERVAL = 10          # 行情轮询间隔（秒）
SNAPSHOT_INTERVAL = 300     # 快照写入间隔（秒，5分钟）
MARKET_OPEN = (9, 36)       # 监控开始（避开Phase B的09:35）
MARKET_CLOSE = (15, 0)      # 监控结束
LUNCH_START = (11, 30)      # 午休开始
LUNCH_END = (13, 0)         # 午休结束


class RealtimeMonitor:
    
    def __init__(self):
        self.engine = storage._get_engine()
        self.rm = RobotManager()
        self._last_snapshot_time = {}   # {stock_code: last_snapshot_timestamp}
        self._running = False
    
    def start(self):
        """启动监控主循环"""
        self._running = True
        logger.info("🟢 实时监控启动")
        
        while self._running:
            now = datetime.now()
            
            # 检查是否在交易时段
            if not self._is_trading_time(now):
                time.sleep(30)
                continue
            
            try:
                self._run_one_cycle(now)
            except Exception as e:
                logger.error(f"监控循环异常: {e}", exc_info=True)
            
            time.sleep(POLL_INTERVAL)
        
        logger.info("🔴 实时监控已停止")
    
    def stop(self):
        """停止监控"""
        self._running = False
    
    def _is_trading_time(self, now: datetime) -> bool:
        """判断是否在交易时段"""
        t = (now.hour, now.minute)
        
        # 周末不交易
        if now.weekday() >= 5:
            return False
        
        # 上午 9:36 - 11:30
        if MARKET_OPEN <= t < (LUNCH_START[0], LUNCH_START[1]):
            return True
        
        # 下午 13:00 - 15:00
        if (LUNCH_END[0], LUNCH_END[1]) <= t < (MARKET_CLOSE[0], MARKET_CLOSE[1]):
            return True
        
        return False
    
    def _run_one_cycle(self, now: datetime):
        """执行一轮监控"""
        trade_date = now.strftime('%Y%m%d')
        
        # 获取所有运行中机器人的持仓
        robots = self.rm.list_robots(status='running')
        
        # 收集所有需要监控的股票
        holdings_map = {}  # {stock_code: [(robot, holding), ...]}
        
        with self.engine.connect() as conn:
            for robot in robots:
                holdings = conn.execute(text('''
                    SELECT id, stock_code, stock_name, buy_price, buy_shares, trade_date
                    FROM sim_trades
                    WHERE robot_id = :rid AND status = 'HOLDING'
                '''), {'rid': robot['robot_id']}).fetchall()
                
                for h in holdings:
                    h = dict(h._mapping)
                    code = h['stock_code']
                    if code not in holdings_map:
                        holdings_map[code] = []
                    holdings_map[code].append((robot, h))
        
        if not holdings_map:
            return
        
        # 批量获取实时行情
        from skills.eastmoney_api import EastMoneyAPI
        api = EastMoneyAPI()
        
        for code, robot_holdings in holdings_map.items():
            try:
                quote = api.realtime(code)
                if not quote or not quote.get('price'):
                    continue
                
                # 写入快照（每5分钟一次）
                self._maybe_save_snapshot(code, quote, trade_date, now)
                
                # 对每个持有该股票的机器人检查信号
                for robot, holding in robot_holdings:
                    self._check_signals(robot, holding, quote, trade_date, now)
                    
            except Exception as e:
                logger.warning(f"监控 {code} 异常: {e}")
    
    def _maybe_save_snapshot(self, code: str, quote: dict, trade_date: str, now: datetime):
        """每5分钟存一条快照"""
        last_time = self._last_snapshot_time.get(code)
        
        if last_time and (now - last_time).total_seconds() < SNAPSHOT_INTERVAL:
            return
        
        with self.engine.begin() as conn:
            conn.execute(text('''
                INSERT INTO market_snapshots
                (stock_code, stock_name, trade_date, snapshot_time,
                 price, open, high, low, pre_close,
                 volume, amount, change_pct, turnover, interval_type)
                VALUES
                (:code, :name, :date, :time,
                 :price, :open, :high, :low, :pre_close,
                 :volume, :amount, :change_pct, :turnover, '5min')
            '''), {
                'code': code,
                'name': quote.get('name'),
                'date': trade_date,
                'time': now,
                'price': quote.get('price'),
                'open': quote.get('open'),
                'high': quote.get('high'),
                'low': quote.get('low'),
                'pre_close': quote.get('pre_close'),
                'volume': quote.get('volume'),
                'amount': quote.get('amount'),
                'change_pct': quote.get('change_pct'),
                'turnover': quote.get('turnover'),
            })
        
        self._last_snapshot_time[code] = now
    
    def _check_signals(self, robot: dict, holding: dict, quote: dict, 
                       trade_date: str, now: datetime):
        """检查单个持仓的交易信号"""
        robot_id = robot['robot_id']
        code = holding['stock_code']
        price = quote.get('price', 0)
        buy_price = holding.get('buy_price', 0)
        
        if not price or not buy_price:
            return
        
        profit_pct = (price - buy_price) / buy_price
        
        # 加载策略
        strategy = load_strategy(
            robot['execute_strategy'],
            robot.get('strategy_params', {})
        )
        
        # 策略判断是否卖出
        signal = strategy.should_sell(holding, quote, {'robot': robot})
        
        if signal is None:
            return
        
        # 记录监控事件
        event_type = 'STOP_LOSS' if '止损' in signal.reason else \
                     'STOP_PROFIT' if '止盈' in signal.reason else 'SIGNAL'
        
        logger.warning(
            f"🚨 {robot['robot_name']} | {code} {holding['stock_name']} | "
            f"{signal.reason} | 当前{profit_pct:+.2%}"
        )
        
        with self.engine.begin() as conn:
            # 执行卖出
            profit = execute_sell(
                pos=holding,
                raw_price=price,
                trade_date=trade_date,
                conn=conn,
                reason=signal.reason,
            )
            
            # 卖出到账
            sell_price = calc_slippage_price(price, 'SELL')
            sell_cost = calc_sell_cost(sell_price, holding['buy_shares'], code)
            
            cm = CashManager(robot_id)
            cm.credit_sell(
                total_received=sell_cost['total_received'],
                trade_id=holding['id'],
                stock_code=code,
                stock_name=holding['stock_name'],
                trade_date=trade_date,
                detail=signal.reason,
            )
            
            # 记录监控事件
            conn.execute(text('''
                INSERT INTO monitor_events
                (robot_id, stock_code, stock_name, trade_date, event_time,
                 event_type, trigger_price, threshold, current_pct,
                 action_taken, trade_id, memo)
                VALUES
                (:rid, :code, :name, :date, :time,
                 :type, :trigger, :threshold, :pct,
                 true, :tid, :memo)
            '''), {
                'rid': robot_id,
                'code': code,
                'name': holding['stock_name'],
                'date': trade_date,
                'time': now,
                'type': event_type,
                'trigger': price,
                'threshold': robot.get('stop_loss_pct', -0.5),
                'pct': round(profit_pct * 100, 2),
                'tid': holding['id'],
                'memo': signal.reason,
            })
        
        logger.info(
            f"✅ 已执行{event_type}: {code} {holding['stock_name']}, "
            f"净利润 ¥{profit['net_profit']:.2f}"
        )
    
    def _check_batch_buy(self, robot: dict, holding: dict, quote: dict,
                         trade_date: str, now: datetime):
        """检查分批建仓信号"""
        if not robot.get('enable_batch_buy'):
            return
        
        strategy = load_strategy(
            robot['execute_strategy'],
            robot.get('strategy_params', {})
        )
        
        signal = strategy.on_batch_buy(holding, quote, {'robot': robot})
        
        if signal is None:
            return
        
        logger.info(
            f"📈 分批加仓信号: {robot['robot_name']} | "
            f"{holding['stock_code']} | {signal.reason}"
        )
        
        # 执行加仓（复用 Phase B 买入逻辑）
        from .position_calc import calculate_buy_shares
        
        calc = calculate_buy_shares(
            planned_amount=float(holding.get('buy_amount', 0)) / signal.batch_total,
            raw_price=quote['price'],
            stock_code=holding['stock_code'],
            total_capital=robot['initial_capital'],
        )
        
        if not calc['valid']:
            logger.warning(f"加仓计算失败: {calc['reason']}")

        cm = CashManager(robot['robot_id'])

            trade_date=trade_date,
            detail=f"分批加仓 第{signal.batch_index}/{signal.batch_total}批",
        )

                # 更新持仓数量和均价

                new_avg_price = (
                    holding['buy_price'] * holding['buy_shares'] +
                    calc['slippage_price'] * calc['shares']
                ) / new_shares
                
                conn.execute(text('''
                    UPDATE sim_trades
                    SET buy_shares = :shares, buy_price = :price, updated_at = NOW()
                    WHERE id = :id
                '''), {
                    'shares': new_shares,
                    'price': round(new_avg_price, 2),
                    'id': holding['id'],
                })
                
                # 写入事件
                conn.execute(text('''
                    INSERT INTO sim_trade_events
                    (trade_id, stock_code, stock_name, direction, trade_date,
                     price, shares, amount, commission, transfer_fee,
```

### 二、淘汰引擎

```python
# skills/stock_sim_robot/eliminator.py
"""
机器人淘汰引擎

流程：
1. 检查是否到达淘汰日
2. 计算所有运行中机器人评分
3. 淘汰得分最低的N个
4. 强制清仓 + 状态改为 eliminated
5. 记录同期所有机器人快照
"""
import json
import logging
from datetime import datetime
from utils.storage import storage
from sqlalchemy import text
from .robot_manager import RobotManager
from .scorer import calc_robot_score, rank_robots

logger = logging.getLogger(__name__)


class Eliminator:
    
    def __init__(self):
        self.engine = storage._get_engine()
        self.rm = RobotManager()
    
    def check_and_run(self, trade_date: str = None):
        """
        检查是否需要执行淘汰，如果需要则执行
        建议每日结算后调用
        """
        if trade_date is None:
            trade_date = datetime.now().strftime('%Y%m%d')
        
        with self.engine.connect() as conn:
            schedules = conn.execute(text('''
                SELECT * FROM elimination_schedule WHERE is_active = true
            ''')).fetchall()
        
        for schedule in schedules:
            schedule = dict(schedule._mapping)
            
            if self._should_run(schedule, trade_date):
                self._execute_elimination(schedule, trade_date)
    
    def _should_run(self, schedule: dict, trade_date: str) -> bool:
        """判断是否到达淘汰日"""
        last_run = schedule.get('last_run_date')
        
        if not last_run:
            return True
        
        # 计算距上次淘汰过了多少交易日
        with self.engine.connect() as conn:
            trading_days = conn.execute(text('''
                SELECT COUNT(DISTINCT trade_date)
                FROM sim_daily_summary
                WHERE trade_date > :last AND trade_date <= :current
            '''), {'last': last_run, 'current': trade_date}).scalar()
        
        return trading_days >= schedule['period_days']
    
    def _execute_elimination(self, schedule: dict, trade_date: str):
        """执行淘汰"""
        logger.info(f"=== 淘汰检查开始: {trade_date} ===")
        
        # 获取运行中机器人
        running = self.rm.list_robots(status='running')
        running_ids = [r['robot_id'] for r in running]
        
        # 最低机器人数量保护
        if len(running_ids) <= schedule['min_robots']:
            logger.info(
                f"运行中机器人 {len(running_ids)} 个，"
                f"不超过最低保留数 {schedule['min_robots']}，跳过淘汰"
            )
            return
        
        # 加载评分权重
        weights = None
        weight_config = schedule.get('weight_config', 'default')
        with self.engine.connect() as conn:
            row = conn.execute(text('''
                SELECT weights FROM score_weights_config
                WHERE config_name = :name AND is_active = true
            '''), {'name': weight_config}).fetchone()
            if row:
                weights = json.loads(row[0]) if isinstance(row[0], str) else row[0]
        
        # 排名
        rankings = rank_robots(running_ids, weights)
        
        if not rankings:
            logger.warning("无法计算排名，跳过淘汰")
            return
        
        # 记录同期所有机器人快照（淘汰前）
        peer_snapshot = []
        for r in rankings:
            robot_info = next((rb for rb in running if rb['robot_id'] == r['robot_id']), {})
            peer_snapshot.append({
                'robot_id': r['robot_id'],
                'robot_name': robot_info.get('robot_name', ''),
                'rank': r['rank'],
                'total_score': r['total_score'],
                'metrics': r['metrics'],
                'scores': r['scores'],
            })
        
        logger.info("当前排名:")
        for p in peer_snapshot:
            logger.info(
                f"  #{p['rank']} {p['robot_name']} "
                f"({p['robot_id']}): {p['total_score']:.1f}分"
            )
        
        # 淘汰最低分的N个
        eliminate_count = min(schedule['eliminate_count'], len(rankings) - schedule['min_robots'])
        to_eliminate = rankings[-eliminate_count:]
        
        for victim in to_eliminate:
            victim_id = victim['robot_id']
            victim_info = next((rb for rb in running if rb['robot_id'] == victim_id), {})
            victim_name = victim_info.get('robot_name', victim_id)
            
            logger.warning(
                f"🗑️ 淘汰: {victim_name} ({victim_id}), "
                f"评分={victim['total_score']:.1f}, "
                f"排名={victim['rank']}/{len(rankings)}"
            )
            
            # 强制清仓并结束
            self.rm.end_robot(victim_id, trade_date)
            
            # 更新状态为 eliminated
            with self.engine.begin() as conn:
                conn.execute(text('''
                    UPDATE sim_robots
                    SET status = 'eliminated',
                        eliminate_reason = :reason,
                        final_score = :score,
                        updated_at = NOW()
                    WHERE robot_id = :rid
                '''), {
                    'rid': victim_id,
                    'reason': f"定期淘汰: 排名{victim['rank']}/{len(rankings)}, "
                              f"评分{victim['total_score']:.1f}",
                    'score': victim['total_score'],
                })
                
                # 写入淘汰记录
                conn.execute(text('''
                    INSERT INTO elimination_records
                    (robot_id, robot_name, eliminate_date,
                     final_score, final_metrics, peer_snapshot, reason)
                    VALUES
                    (:rid, :name, :date,
                     :score, :metrics::jsonb, :peers::jsonb, :reason)
                '''), {
                    'rid': victim_id,
                    'name': victim_name,
                    'date': trade_date,
                    'score': victim['total_score'],
                    'metrics': json.dumps(victim['metrics'], ensure_ascii=False),
                    'peers': json.dumps(peer_snapshot, ensure_ascii=False),
                    'reason': f"排名{victim['rank']}/{len(rankings)}",
                })
        
        # 更新淘汰计划执行日期
        with self.engine.begin() as conn:
            conn.execute(text('''
                UPDATE elimination_schedule
                SET last_run_date = :date
                WHERE id = :sid
            '''), {'date': trade_date, 'sid': schedule['id']})
        
        logger.info(f"=== 淘汰完成: 淘汰 {len(to_eliminate)} 个 ===")
    
    def get_elimination_history(self) -> list[dict]:
        """查询淘汰历史"""
        with self.engine.connect() as conn:
            rows = conn.execute(text('''
                SELECT id, robot_id, robot_name, eliminate_date,
                       final_score, final_metrics, peer_snapshot, reason
                FROM elimination_records
                ORDER BY eliminate_date DESC
            ''')).fetchall()
        
        return [dict(r._mapping) for r in rows]
    
    def get_elimination_detail(self, record_id: int) -> dict:
        """查询单次淘汰详情（含同期对比）"""
        with self.engine.connect() as conn:
            row = conn.execute(text('''
                SELECT * FROM elimination_records WHERE id = :id
            '''), {'id': record_id}).fetchone()
        
        if not row:
            return None
        
        result = dict(row._mapping)
        # 解析JSON
        if isinstance(result['peer_snapshot'], str):
            result['peer_snapshot'] = json.loads(result['peer_snapshot'])
        if isinstance(result['final_metrics'], str):
            result['final_metrics'] = json.loads(result['final_metrics'])
        
        return result
```

### 三、除权分红处理

```python
# skills/stock_sim_robot/dividend_handler.py
"""
除权分红处理

流程：
1. 每日盘前检查持仓股票是否有除权除息
2. 如果有分红 → 现金到账
3. 如果有送股/转增 → 增加持仓股数，调整均价（前复权）
4. 数据源：东方财富API

调用时机：每日09:00（Phase B之前）
"""
import logging
from datetime import datetime
from utils.storage import storage
from sqlalchemy import text
from .robot_manager import RobotManager
from .cash_manager import CashManager

logger = logging.getLogger(__name__)


class DividendHandler:
    
    def __init__(self):
        self.engine = storage._get_engine()
        self.rm = RobotManager()
    
    def run(self, trade_date: str = None):
        """检查并处理所有机器人的除权分红"""
        if trade_date is None:
            trade_date = datetime.now().strftime('%Y%m%d')
        
        logger.info(f"=== 除权分红检查: {trade_date} ===")
        
        # 获取今天除权的股票列表
        ex_stocks = self._fetch_ex_dividend_stocks(trade_date)
        
        if not ex_stocks:
            logger.info("今日无除权除息股票")
            return
        
        logger.info(f"今日除权除息: {len(ex_stocks)} 只")
        
        # 检查每个机器人的持仓
        robots = self.rm.list_robots(status='running')
        
        for robot in robots:
            self._process_robot(robot, ex_stocks, trade_date)
        
        logger.info(f"=== 除权分红处理完成 ===")
    
    def _fetch_ex_dividend_stocks(self, trade_date: str) -> dict:
        """
        获取今日除权除息的股票
        
        Returns:
            {stock_code: {
                'dividend_per_share': 每股分红,
                'bonus_ratio': 每股送股比例（如10送2则为0.2）,
                'split_ratio': 每股转增比例,
            }}
        """
        # 先查本地缓存
        with self.engine.connect() as conn:
            cached = conn.execute(text('''
                SELECT stock_code, dividend_per_share, bonus_ratio, split_ratio
                FROM dividend_records
                WHERE ex_date = :date
            '''), {'date': trade_date}).fetchall()
            
            if cached:
                return {
                    r[0]: {
                        'dividend_per_share': r[1],
                        'bonus_ratio': r[2],
                        'split_ratio': r[3],
                    }
                    for r in cached
                }
        
        # 没有缓存则从API获取
        try:
            result = self._fetch_from_api(trade_date)
            
            # 写入缓存
            if result:
                with self.engine.begin() as conn:
                    for code, info in result.items():
                        conn.execute(text('''
                            INSERT INTO dividend_records
                            (stock_code, ex_date, dividend_per_share, 
                             bonus_ratio, split_ratio, source)
                            VALUES (:code, :date, :div, :bonus, :split, 'eastmoney')
                            ON CONFLICT (stock_code, ex_date) DO NOTHING
                        '''), {
                            'code': code,
                            'date': trade_date,
                            'div': info['dividend_per_share'],
                            'bonus': info['bonus_ratio'],
                            'split': info['split_ratio'],
                        })
            
            return result
            
        except Exception as e:
            logger.error(f"获取除权数据失败: {e}")
            return {}
    
    def _fetch_from_api(self, trade_date: str) -> dict:
        """从东方财富API获取除权除息数据"""
        from skills.eastmoney_api import EastMoneyAPI
        api = EastMoneyAPI()
        
        # 东方财富分红送配接口
        # 注意：实际接口可能需要调整
        try:
            import requests
            url = "https://datacenter-web.eastmoney.com/api/data/v1/get"
            params = {
                "sortColumns": "EX_DIVIDEND_DATE",
                "sortTypes": -1,
                "pageSize": 500,
                "pageNumber": 1,
                "reportName": "RPT_SHAREBONUS_DET",
                "columns": "ALL",
                "filter": f'(EX_DIVIDEND_DATE=\'{trade_date[:4]}-{trade_date[4:6]}-{trade_date[6:]}\')',
            }
            
            resp = requests.get(url, params=params, timeout=10)
            data = resp.json()
            
            result = {}
            items = data.get('result', {}).get('data', [])
            
            for item in items:
                code = item.get('SECURITY_CODE', '')
                if not code:
                    continue
                
                result[code] = {
                    'dividend_per_share': float(item.get('BONUS_IT_RATIO', 0) or 0) / 10,
                    'bonus_ratio': float(item.get('BONUS_SH_RATIO', 0) or 0) / 10,
                    'split_ratio': float(item.get('CONVERT_SH_RATIO', 0) or 0) / 10,
                    'stock_name': item.get('SECURITY_NAME_ABBR', ''),
                }
            
            return result
            
        except Exception as e:
            logger.warning(f"API获取分红数据异常: {e}")
            return {}
    
    def _process_robot(self, robot: dict, ex_stocks: dict, trade_date: str):
        """处理单个机器人的除权分红"""
        robot_id = robot['robot_id']
        
        with self.engine.begin() as conn:
            holdings = conn.execute(text('''
                SELECT id, stock_code, stock_name, buy_price, buy_shares
                FROM sim_trades
                WHERE robot_id = :rid AND status = 'HOLDING'
            '''), {'rid': robot_id}).fetchall()
            
            for h in holdings:
                h = dict(h._mapping)
                code = h['stock_code']
                
                if code not in ex_stocks:
                    continue
                
                ex_info = ex_stocks[code]
                dividend = ex_info['dividend_per_share']
                bonus = ex_info['bonus_ratio']
                split = ex_info['split_ratio']
                
                logger.info(
                    f"📌 {robot['robot_name']} | {code} {h['stock_name']} | "
                    f"分红={dividend}元/股, 送股={bonus}, 转增={split}"
                )
                
                # 处理现金分红
                if dividend > 0:
                    cash_amount = dividend * h['buy_shares']
                    cm = CashManager(robot_id)
                    cm.credit_dividend(
                        amount=round(cash_amount, 2),
                        stock_code=code,
                        stock_name=h['stock_name'],
                        trade_date=trade_date,
                    )
                    logger.info(f"  💰 分红到账: ¥{cash_amount:.2f}")
                
                # 处理送股/转增
                total_bonus_ratio = bonus + split
                if total_bonus_ratio > 0:
                    old_shares = h['buy_shares']
                    new_shares = int(old_shares * (1 + total_bonus_ratio))
                    # 确保是100的整数倍
                    new_shares = (new_shares // 100) * 100
                    if new_shares < old_shares:
                        new_shares = old_shares
                    
                    added_shares = new_shares - old_shares
                    
                    # 前复权：调整买入均价
                    # 前复权价 = 原价 / (1 + 送转比例)
                    old_price = h['buy_price']
                    new_price = round(old_price / (1 + total_bonus_ratio), 2)
                    
                    conn.execute(text('''
                        UPDATE sim_trades
                        SET buy_shares = :shares,
                            buy_price = :price,
                            updated_at = NOW()
                        WHERE id = :id
                    '''), {
                        'shares': new_shares,
                        'price': new_price,
                        'id': h['id'],
                    })
                    
                    logger.info(
                        f"  📈 送转股: {old_shares}→{new_shares}股(+{added_shares}), "
                        f"均价 {old_price}→{new_price}(前复权)"
                    )
```

```python
# scripts/daily_cron.py
"""
每日调度主入口

crontab 配置：
0  9  * * 1-5  cd ~/stock_analysis && python scripts/daily_cron.py dividend
35 9  * * 1-5  cd ~/stock_analysis && python scripts/daily_cron.py phase_b

import sys
import logging
from datetime import datetime

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(name)s] %(levelname)s: %(message)s',
    handlers=[
        logging.FileHandler(f'logs/cron_{datetime.now().strftime("%Y%m%d")}.log'),
        logging.StreamHandler(),
    ]
)
logger = logging.getLogger('daily_cron')


def main():
    if len(sys.argv) < 2:
        print("用法: python daily_cron.py <command>")
        print("命令: dividend | phase_b | phase_a | settle | eliminate | backup")
        sys.exit(1)
    
    command = sys.argv[1]
    trade_date = sys.argv[2] if len(sys.argv) > 2 else datetime.now().strftime('%Y%m%d')
    
    logger.info(f"=== daily_cron: {command} | {trade_date} ===")
    
    try:
        if command == 'dividend':
            # 09:00 - 除权分红处理（Phase B 之前）
            from skills.stock_sim_robot.dividend_handler import DividendHandler
            DividendHandler().run(trade_date)
        
        elif command == 'phase_b':
            # 09:35 - 开盘买入
            from skills.stock_sim_robot.trade_runner import run_phase_b
            run_phase_b(trade_date)
        
        elif command == 'phase_a':
            # 18:45 - 盘后选股
            from skills.stock_sim_robot.trade_runner import run_phase_a
            run_phase_a(trade_date)
        
        elif command == 'settle':
            # 15:05 - 每日结算 + 淘汰检查
            from skills.stock_sim_robot.daily_settlement import DailySettlement
            DailySettlement().run(trade_date)
            
            # 结算后执行淘汰检查
            from skills.stock_sim_robot.eliminator import Eliminator
            Eliminator().check_and_run(trade_date)
        
        elif command == 'eliminate':
            # 手动触发淘汰
            from skills.stock_sim_robot.eliminator import Eliminator
            Eliminator().check_and_run(trade_date)
        
        elif command == 'backup':
            # 数据库备份
            import subprocess
            subprocess.run(['bash', 'scripts/backup_db.sh'], check=True)
        
        elif command == 'verify':
            # 全量验证
            from scripts.verify_p0 import verify as verify_p0
            verify_p0()
        
        else:
            logger.error(f"未知命令: {command}")
            sys.exit(1)
        
        logger.info(f"=== {command} 执行完成 ===")
    
    except Exception as e:
        logger.error(f"❌ {command} 执行失败: {e}", exc_info=True)
        sys.exit(1)


if __name__ == '__main__':
    main()
```

### 六、历史数据清理

```python
# skills/stock_sim_robot/data_cleaner.py
"""
历史数据清理（节省磁盘空间）

清理策略：
- market_snapshots: 保留最近90天，更早的只保留日级别
- monitor_events: 保留最近180天
- 其他表: 不清理（数据量小）
"""
import logging
from datetime import datetime, timedelta
from utils.storage import storage
from sqlalchemy import text

logger = logging.getLogger(__name__)


def cleanup(keep_days_snapshots: int = 90, keep_days_events: int = 180):
    """执行数据清理"""
    engine = storage._get_engine()
    now = datetime.now()
    
    with engine.begin() as conn:
        
        # === 1. 清理行情快照 ===
        cutoff_snap = (now - timedelta(days=keep_days_snapshots)).strftime('%Y%m%d')
        
        # 保留每天最后一条（收盘价），删除其余
        deleted = conn.execute(text('''
            DELETE FROM market_snapshots
            WHERE trade_date < :cutoff
              AND id NOT IN (
                  SELECT DISTINCT ON (stock_code, trade_date) id
                  FROM market_snapshots
                  WHERE trade_date < :cutoff
                  ORDER BY stock_code, trade_date, snapshot_time DESC
              )
        '''), {'cutoff': cutoff_snap})
        
        logger.info(f"📦 行情快照: 清理 {cutoff_snap} 之前的盘中数据, 删除 {deleted.rowcount} 条")
        
        # === 2. 清理监控事件 ===
        cutoff_evt = (now - timedelta(days=keep_days_events)).strftime('%Y%m%d')
        
        deleted = conn.execute(text('''
            DELETE FROM monitor_events WHERE trade_date < :cutoff
        '''), {'cutoff': cutoff_evt})
        
        logger.info(f"📦 监控事件: 清理 {cutoff_evt} 之前, 删除 {deleted.rowcount} 条")
        
        # === 3. 统计当前数据量 ===
        tables = [
            'sim_robots', 'sim_trades', 'sim_trade_events',
            'cash_ledger', 'sim_daily_summary', 'market_snapshots',
            'monitor_events', 'benchmark_daily', 'elimination_records',
        ]
        
        logger.info("\n📊 数据量统计:")
        for table in tables:
            try:
                count = conn.execute(text(f'SELECT COUNT(*) FROM {table}')).scalar()
                size = conn.execute(text(f'''
                    SELECT pg_size_pretty(pg_total_relation_size('{table}'))
                ''')).scalar()
                logger.info(f"  {table}: {count} 条, {size}")
            except Exception:
                pass


if __name__ == '__main__':
    cleanup()
```
